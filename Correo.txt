@ConfigurationProperties(prefix = "clients.document")
public record DocumentApiProperties(
        String baseUrl,
        String clientId
) {
}


@ConfigurationProperties(prefix = "clients.signature")
public record SignatureApiProperties(
        String baseUrl,
        String clientId
) {
}

@Configuration
@EnableConfigurationProperties({
        DocumentApiProperties.class,
        SignatureApiProperties.class
})
public class WebClientConfig {

    @Bean
    public WebClient documentWebClient(
            WebClient.Builder builder,
            DocumentApiProperties properties) {

        return builder
                .baseUrl(properties.baseUrl())
                .defaultHeader(
                        "X-Client-Id",
                        properties.clientId()
                )
                .build();
    }

    @Bean
    public WebClient signatureWebClient(
            WebClient.Builder builder,
            SignatureApiProperties properties) {

        return builder
                .baseUrl(properties.baseUrl())
                .defaultHeader(
                        "X-Client-Id",
                        properties.clientId()
                )
                .build();
    }
}



@Component
@RequiredArgsConstructor
public class DocumentClient {

    private final WebClient documentWebClient;

    public Mono<DocumentApiResponse> getDocument(
            String documentId) {

        return documentWebClient
                .get()
                .uri("/documents/{documentId}", documentId)
                .retrieve()
                .bodyToMono(DocumentApiResponse.class);
    }
}




@Component
@RequiredArgsConstructor
public class SignatureClient {

    private final WebClient signatureWebClient;

    public Mono<SignatureApiResponse> signDocument(
            SignatureRequest request) {

        return signatureWebClient
                .post()
                .uri("/signatures")
                .bodyValue(request)
                .retrieve()
                .bodyToMono(SignatureApiResponse.class);
    }
}



public class DocumentApiException extends RuntimeException {

    private final String errorCode;

    public DocumentApiException(
            String errorCode,
            String message) {

        super(message);
        this.errorCode = errorCode;
    }

    public String getErrorCode() {
        return errorCode;
    }
}



public class SignatureApiException extends RuntimeException {

    private final String errorCode;

    public SignatureApiException(
            String errorCode,
            String message) {

        super(message);
        this.errorCode = errorCode;
    }

    public String getErrorCode() {
        return errorCode;
    }
}



public Mono<DocumentApiResponse> getDocument(String documentId) {

    return documentWebClient
            .get()
            .uri("/documents/{documentId}", documentId)
            .retrieve()
            .onStatus(
                    status -> status.is4xxClientError(),
                    response -> response
                            .bodyToMono(String.class)
                            .defaultIfEmpty("")
                            .flatMap(body ->
                                    Mono.error(
                                            new DocumentApiException(
                                                    "DOCUMENT_API_4XX",
                                                    body
                                            )
                                    )
                            )
            )
            .onStatus(
                    status -> status.is5xxServerError(),
                    response -> response
                            .bodyToMono(String.class)
                            .defaultIfEmpty("")
                            .flatMap(body ->
                                    Mono.error(
                                            new DocumentApiException(
                                                    "DOCUMENT_API_5XX",
                                                    body
                                            )
                                    )
                            )
            )
            .bodyToMono(DocumentApiResponse.class);
}


@Service
@RequiredArgsConstructor
public class DocumentProcessService {

    private static final int MAX_CONCURRENCY = 5;

    private final DocumentClient documentClient;
    private final SignatureClient signatureClient;
    private final DocumentErrorService errorService;

    public Mono<DocumentProcessResponse> process(
            DocumentProcessRequest request) {

        return Flux.fromIterable(request.documentIds())
                .flatMap(
                        documentId ->
                                processDocument(
                                        documentId,
                                        request.client()
                                ),
                        MAX_CONCURRENCY
                )
                .collectList()
                .map(this::buildResponse);
    }

private Mono<DocumentResult> processDocument(
        String documentId,
        ClientData client) {

    return documentClient
            .getDocument(documentId)
            .flatMap(document ->
                    signDocument(
                            documentId,
                            document,
                            client
                    )
            )
            .thenReturn(
                    DocumentResult.success(documentId)
            )
            .onErrorResume(error ->
                    handleDocumentError(
                            documentId,
                            error
                    )
            );
}


private Mono<SignatureApiResponse> signDocument(
        String documentId,
        DocumentApiResponse document,
        ClientData client) {

    SignatureRequest request =
            buildSignatureRequest(
                    documentId,
                    document,
                    client
            );

    return signatureClient.signDocument(request);
}

private SignatureRequest buildSignatureRequest(
        String documentId,
        DocumentApiResponse document,
        ClientData client) {

    SignerData signer = new SignerData(
            client.documentType(),
            client.documentNumber(),
            client.name(),
            client.lastName(),
            client.email()
    );

    return new SignatureRequest(
            documentId,
            document.fileBase64(),
            signer
    );
}

private Mono<DocumentApiResponse> retrieveDocument(
        String documentId) {

    return documentClient
            .getDocument(documentId)
            .onErrorMap(error ->
                    new DocumentProcessException(
                            documentId,
                            ErrorStage.DOCUMENT_RETRIEVAL,
                            extractErrorCode(error),
                            error.getMessage(),
                            error
                    )
            );
}


private Mono<SignatureApiResponse> signDocument(
        String documentId,
        DocumentApiResponse document,
        ClientData client) {

    SignatureRequest request =
            buildSignatureRequest(
                    documentId,
                    document,
                    client
            );

    return signatureClient
            .signDocument(request)
            .onErrorMap(error ->
                    new DocumentProcessException(
                            documentId,
                            ErrorStage.SIGNATURE,
                            extractErrorCode(error),
                            error.getMessage(),
                            error
                    )
            );
}


@Getter
public class DocumentProcessException
        extends RuntimeException {

    private final String documentId;
    private final ErrorStage stage;
    private final String errorCode;

    public DocumentProcessException(
            String documentId,
            ErrorStage stage,
            String errorCode,
            String message,
            Throwable cause) {

        super(message, cause);

        this.documentId = documentId;
        this.stage = stage;
        this.errorCode = errorCode;
    }
}


private Mono<DocumentResult> processDocument(
        String documentId,
        ClientData client) {

    return retrieveDocument(documentId)
            .flatMap(document ->
                    signDocument(
                            documentId,
                            document,
                            client
                    )
            )
            .thenReturn(
                    DocumentResult.success(documentId)
            )
            .onErrorResume(
                    DocumentProcessException.class,
                    this::handleDocumentError
            );
}


private Mono<DocumentResult> handleDocumentError(
        DocumentProcessException error) {

    return errorService
            .save(error)
            .thenReturn(
                    DocumentResult.failure(
                            error.getDocumentId(),
                            error.getStage().name(),
                            error.getErrorCode(),
                            error.getMessage()
                    )
            );
}





private DocumentProcessResponse buildResponse(
        List<DocumentResult> results) {

    int total = results.size();

    int successful = (int) results.stream()
            .filter(result ->
                    result.status() == DocumentStatus.SUCCESS)
            .count();

    int failed = total - successful;

    return new DocumentProcessResponse(
            total,
            successful,
            failed,
            results
    );
}






@Service
@RequiredArgsConstructor
public class DocumentErrorService {

    private final DocumentErrorRepository repository;

    public Mono<Void> save(
            DocumentProcessException error) {

        DocumentErrorEntity entity =
                new DocumentErrorEntity(
                        error.getDocumentId(),
                        error.getStage().name(),
                        error.getErrorCode(),
                        error.getMessage()
                );

        return repository.save(entity)
                .then();
    }
}
