import { DynamoDBClient, GetItemCommand } from "@aws-sdk/client-dynamodb";
import { unmarshall } from "@aws-sdk/util-dynamodb";
import jwt from "jsonwebtoken";

// Cliente DynamoDB optimizado (reutilizado en invocaciones)
const ddbClient = new DynamoDBClient({ region: process.env.AWS_REGION });
const TABLE_NAME = process.env.TOKEN_TABLE; // Ej: "JWTTokens"
const JWT_SECRET = process.env.JWT_SECRET;

export const handler = async (event) => {
  try {
    // 1. Extraer token del header "Authorization: Bearer <token>"
    const token = extractToken(event.authorizationToken);
    if (!token) {
      return generatePolicy("user", "Deny", event.methodArn);
    }

    // 2. Verificar firma y decodificar payload (valida exp automáticamente)
    let decoded;
    try {
      decoded = jwt.verify(token, JWT_SECRET, { algorithms: ["HS256"] });
    } catch (err) {
      console.error("JWT Verification failed:", err.message);
      return generatePolicy("user", "Deny", event.methodArn);
    }

    // 3. Validar que el payload CONTENGA el campo 'jti'
    const { jti, sub } = decoded;
    if (!jti) {
      console.warn("Token missing 'jti' claim");
      return generatePolicy("user", "Deny", event.methodArn);
    }

    // 4. Consultar DynamoDB usando el jti como clave primaria
    const dbItem = await getTokenFromDB(jti);
    if (!dbItem) {
      console.warn(`Token jti ${jti} not found in DynamoDB`);
      return generatePolicy("user", "Deny", event.methodArn);
    }

    // 5. Verificar estado en base de datos (ej: "ACTIVE" o "REVOKED")
    if (dbItem.status !== "ACTIVE") {
      console.warn(`Token jti ${jti} is ${dbItem.status}`);
      return generatePolicy("user", "Deny", event.methodArn);
    }

    // 6. (Opcional) Validar expiración adicional en DB o comparar fechas
    // Si todo ok → PERMITIR acceso
    console.info(`Access granted for user ${sub} with jti ${jti}`);
    return generatePolicy(sub, "Allow", event.methodArn, {
      // Contexto que pasa al backend (útil para logging)
      userId: sub,
      tokenId: jti,
    });
  } catch (error) {
    console.error("Unexpected error:", error);
    return generatePolicy("user", "Deny", event.methodArn);
  }
};

// --- Funciones auxiliares ---

function extractToken(authHeader) {
  if (!authHeader) return null;
  const parts = authHeader.split(" ");
  if (parts.length === 2 && parts[0] === "Bearer") {
    return parts[1];
  }
  return null;
}

async function getTokenFromDB(jti) {
  const command = new GetItemCommand({
    TableName: TABLE_NAME,
    Key: {
      jti: { S: jti }, // La tabla debe tener 'jti' como Partition Key (String)
    },
    ConsistentRead: false, // Lectura eventual para menor costo/latencia
  });

  const response = await ddbClient.send(command);
  if (!response.Item) return null;

  // Desmapear el item de DynamoDB a objeto JS
  const item = unmarshall(response.Item);
  return item;
}

// Genera la política IAM que API Gateway necesita
function generatePolicy(principalId, effect, resource, context = {}) {
  // Si es Deny, es recomendable negar todo el ARN (o el específico)
  // Para Allow, dar acceso al recurso específico de la petición.
  const policy = {
    principalId: principalId || "user",
    policyDocument: {
      Version: "2012-10-17",
      Statement: [
        {
          Action: "execute-api:Invoke",
          Effect: effect,
          Resource: effect === "Allow" ? resource : "*", // Deny todo o especificar
        },
      ],
    },
    context: context, // Se pasa al Lambda integrado
  };

  // Añadir caché (TTL en segundos) para que API Gateway no llame al authorizer en cada request
  // Se recomienda entre 300 y 3600 segundos según tu caso.
  policy.context = {
    ...context,
    // Opcional: incluir tiempo de expiración del token para cachear hasta ese momento
  };
  // Para activar el cache, debes habilitarlo en la consola de API Gateway (Authorizer Cache TTL)
  // El Lambda devuelve este objeto, API Gateway lo cachea automáticamente si está habilitado.

  return policy;
}
