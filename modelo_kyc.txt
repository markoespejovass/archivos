CREATE TABLE kyc_case (
    id BIGSERIAL PRIMARY KEY,
    party_id VARCHAR(50) NOT NULL,

    structural_segment_code VARCHAR(10),
    structural_segment_desc VARCHAR(100),

    structural_subsegment_code VARCHAR(10),
    structural_subsegment_desc VARCHAR(100),

    preference_channel_code VARCHAR(10),
    preference_channel_desc VARCHAR(100),

    public_office_indicator BOOLEAN,

    creation_date DATE,
    creation_channel_code VARCHAR(10),
    creation_channel_desc VARCHAR(100)
);

CREATE TABLE party (
    party_id VARCHAR(50) PRIMARY KEY,
    kyc_case_id BIGINT,

    status_code VARCHAR(10),
    status_description VARCHAR(100),

    CONSTRAINT fk_party_kyc
        FOREIGN KEY (kyc_case_id)
        REFERENCES kyc_case(id)
);


CREATE TABLE person (
    id BIGSERIAL PRIMARY KEY,
    party_id VARCHAR(50),

    full_name VARCHAR(255),
    given_name VARCHAR(100),
    middle_name VARCHAR(100),
    last_name VARCHAR(100),
    second_last_name VARCHAR(100),

    birth_date DATE,
    gender_description VARCHAR(50),
    civil_status_description VARCHAR(50),

    birth_country_code VARCHAR(10),
    birth_country_name VARCHAR(100),

    is_pep BOOLEAN,
    pep_comments TEXT,

    CONSTRAINT fk_person_party
        FOREIGN KEY (party_id)
        REFERENCES party(party_id)
);


CREATE TABLE person_document (
    id BIGSERIAL PRIMARY KEY,
    person_id BIGINT,
    document_number VARCHAR(50),
    document_type_code VARCHAR(20),

    FOREIGN KEY (person_id) REFERENCES person(id)
);


CREATE TABLE contact_point (
    id BIGSERIAL PRIMARY KEY,
    party_id VARCHAR(50),

    email VARCHAR(150),
    phone VARCHAR(50),

    FOREIGN KEY (party_id) REFERENCES party(party_id)
);

CREATE TABLE employment (
    id BIGSERIAL PRIMARY KEY,
    person_id BIGINT,

    occupation_code VARCHAR(20),
    occupation_description VARCHAR(150),

    status_code VARCHAR(10),
    status_description VARCHAR(100),

    position_code VARCHAR(20),
    position_description VARCHAR(150),

    economic_category_desc VARCHAR(150),
    economic_subcategory_code VARCHAR(20),
    economic_subcategory_desc VARCHAR(150),

    employer_name VARCHAR(150),
    employer_type VARCHAR(100),
    employer_address TEXT,

    is_primary BOOLEAN,

    FOREIGN KEY (person_id) REFERENCES person(id)
);


CREATE TABLE product (
    id BIGSERIAL PRIMARY KEY,
    kyc_case_id BIGINT,

    product_family_id VARCHAR(50),
    product_family_name VARCHAR(150),

    product_code VARCHAR(50),
    product_description VARCHAR(150),

    FOREIGN KEY (kyc_case_id) REFERENCES kyc_case(id)
);


CREATE TABLE questionnaire (
    id BIGSERIAL PRIMARY KEY,
    kyc_case_id BIGINT,

    questionnaire_id VARCHAR(50),
    version VARCHAR(10),

    last_update_date DATE,
    last_update_process VARCHAR(100),
    expiration_date DATE,
    completion_date DATE,
    submission_date DATE,

    FOREIGN KEY (kyc_case_id) REFERENCES kyc_case(id)
);

CREATE TABLE question (
    id BIGSERIAL PRIMARY KEY,
    questionnaire_id BIGINT,

    question_id VARCHAR(50),
    description TEXT,

    type_description VARCHAR(100),
    translation_code VARCHAR(50),

    answer TEXT,
    answer_translation_code VARCHAR(50),

    FOREIGN KEY (questionnaire_id) REFERENCES questionnaire(id)
);


CREATE TABLE question (
    id BIGSERIAL PRIMARY KEY,
    questionnaire_id BIGINT,

    question_id VARCHAR(50),
    description TEXT,

    type_description VARCHAR(100),
    translation_code VARCHAR(50),

    answer TEXT,
    answer_translation_code VARCHAR(50),

    FOREIGN KEY (questionnaire_id) REFERENCES questionnaire(id)
);


CREATE TABLE question_answer_value (
    id BIGSERIAL PRIMARY KEY,
    question_id BIGINT,

    translation_code VARCHAR(50),
    value VARCHAR(100),

    FOREIGN KEY (question_id) REFERENCES question(id)
);




CREATE TABLE kyc_financial_info (
    id BIGSERIAL PRIMARY KEY,
    kyc_case_id BIGINT,

    purpose_of_relationship VARCHAR(100),
    type_of_operations VARCHAR(100),

    main_customer VARCHAR(150),
    main_supplier VARCHAR(150),

    annual_income_amount DECIMAL(18,2),
    annual_income_currency VARCHAR(10),

    patrimony_amount DECIMAL(18,2),
    patrimony_currency VARCHAR(10),

    FOREIGN KEY (kyc_case_id) REFERENCES kyc_case(id)
);



CREATE TABLE source_of_funds (
    id BIGSERIAL PRIMARY KEY,
    kyc_case_id BIGINT,
    code VARCHAR(20),
    description VARCHAR(100),

    type VARCHAR(20), -- WEALTH / FUNDS

    FOREIGN KEY (kyc_case_id) REFERENCES kyc_case(id)
);

CREATE TABLE aml_info (
    id BIGSERIAL PRIMARY KEY,
    kyc_case_id BIGINT,

    aml_id VARCHAR(50),
    status_code VARCHAR(10),

    FOREIGN KEY (kyc_case_id) REFERENCES kyc_case(id)
);

CREATE TABLE aml_risk_category (
    id BIGSERIAL PRIMARY KEY,
    aml_id BIGINT,

    code VARCHAR(20),
    description VARCHAR(150),

    FOREIGN KEY (aml_id) REFERENCES aml_info(id)
);
CREATE TABLE aml_risk_source (
    id BIGSERIAL PRIMARY KEY,
    aml_id BIGINT,

    code VARCHAR(50),

    FOREIGN KEY (aml_id) REFERENCES aml_info(id)
);

CREATE TABLE kyc_analysis (
    id BIGSERIAL PRIMARY KEY,
    kyc_case_id BIGINT,

    analyst_id VARCHAR(50),
    analyst_name VARCHAR(255),
    analyst_email VARCHAR(150),

    is_verified BOOLEAN,

    FOREIGN KEY (kyc_case_id) REFERENCES kyc_case(id)
);
