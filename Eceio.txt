    private WrapperPostDocumentsRequest buildRequest() {

        String base64 = Base64.getEncoder()
                .encodeToString("contenido".getBytes());

        WrapperMySearchDocumentResponseOwnersInner owner =
                new WrapperMySearchDocumentResponseOwnersInner();

        owner.setOwnerId("12345678");

        WrapperMyDocumentRequest document =
                new WrapperMyDocumentRequest();

        document.setFolderReference(base64);
        document.setAclReference("ACL");
        document.setMimeType("application/pdf");
        document.setName("file.pdf");
        document.setOwners(List.of(owner));

        App caller =
                new App();

        caller.setAppId("APP");

        WrapperPostDocumentsRequest request =
                new WrapperPostDocumentsRequest();

        request.setDocument(document);
        request.setCallerInformation(caller);

        return request;
    }
