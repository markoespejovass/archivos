package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.jpa;

import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.input.rest.data.WrapperMySearchFilterRequest;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.input.rest.data.WrapperMySearchFilterRequestSearchPropertiesInner;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils.FileField;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils.FileSearch;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils.Operator;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils.RequestOperator;
import org.apache.logging.log4j.util.Strings;

import java.util.List;

public class FileSearchMapper {

    public static FileSearch toSearch(WrapperMySearchFilterRequest request) {

        FileSearch.Builder builder = FileSearch.builder();

        boolean isOr = "OR".equalsIgnoreCase(request.getDiscriminator());

        for (WrapperMySearchFilterRequestSearchPropertiesInner prop : request.getSearchProperties()) {
            if(prop.getKeyId() == null){
                prop.setKeyId("0");
            }
            String keyId = prop.getKeyId();

            String field = mapField(keyId);
            if(field == null){
                field="";
            }
            WrapperMySearchFilterRequestSearchPropertiesInner.KeyOperatorEnum keyOperator = prop.getKeyOperator();
            if(keyOperator == null){
                keyOperator = WrapperMySearchFilterRequestSearchPropertiesInner.KeyOperatorEnum.IN;
            }
            Operator operator = mapOperator(keyOperator.name());
            Object value = castValue(keyId, prop.getKeyValue());
            if (value == null){
                value = prop.getKeyValue();
            }
            if (isOr) {
                builder.or(field, operator, value);
            } else {
                builder.and(field, operator, value);
            }
        }

        return builder.build();
    }

    private static String mapField(String keyId) {
        try {
            return FileField.valueOf(keyId.toUpperCase()).getDynamoField();
        } catch (Exception e) {
            throw new IllegalArgumentException("Campo no permitido: " + keyId);
        }
    }

    private static Operator mapOperator(String op) {
        try {
            return RequestOperator
                    .valueOf(op.toUpperCase())
                    .toDomain(RequestOperator.valueOf(op.toUpperCase()));
        } catch (Exception e) {
            throw new IllegalArgumentException("Operador no permitido: " + op);
        }
    }

    private static Object castValue(String keyId, String value) {

        // Tipado básico (importante para DynamoDB)
        return switch (keyId.toUpperCase()) {
            case "VERSION" -> Integer.valueOf(value);
            default -> value;
        };
    }
}
