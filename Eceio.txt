@Mapper(componentModel = "spring")
public interface ModelicaMapper {

    @Mapping(source = "documentNumber", target = "documentNumber")
    DocumentIdDTO toDocumentIdDTO(DocumentIdRecord documentIdRecord);

    @Mapping(source = "id", target = "idDTO")
    CustomerAccountHolderDTO toCustomerAccountHolderDTO(CustomerAccountHolder customerAccountHolder);

    @Mapping(source = "customerAccountHolder", target = "customerAccountHolderDTOM")
    ModelicaRequest toModelicaRequest (CreditRiskValidate creditRiskValidate);

    CreditRiskValidateResult toCreditRiskValidateResult (ModelicaResponse modelicaResponse);

}
