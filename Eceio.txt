import software.amazon.awssdk.core.sync.RequestBody;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.*;
import software.amazon.awssdk.services.s3.presigner.S3Presigner;
import software.amazon.awssdk.services.s3.presigner.model.*;

import org.springframework.stereotype.Component;

import java.io.ByteArrayInputStream;
import java.time.Duration;
import java.util.*;

import static com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.constants.Constant.PREFIJO_KEY;
import static com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.util.UtilsFile.*;


/**
 * The class representative Otp adaptarer
 *
 * @author Create by team Proyecto [NUBE] Pe
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class DocumentManagementAdapter implements DocumentManagementRepositoryPort {

  private final S3Client s3Client;

  private final S3Presigner s3Presigner;

  private final DocumentMapper mapper;

  private final FileEntityRepository repository;

  @Value("${aws.bucket}")
  public String bucketName;

  /**
   * Metodo de salida para validar otp
   *
   * @param criteria valor del onjecto request
   * @return WrapperClassifyEmail objecto de response valid
   */
  @Override
  public WrapperGetDocumentsResponse downloadDocument(WrapperGetDocumentsRequest criteria) {
    String documentId;
    String url;
    WrapperMyDocumentResponse documentResponse;
    if(criteria.getDocument() == null){
      return new WrapperGetDocumentsResponse();
    }
    documentResponse = criteria.getDocument();
    if(documentResponse.getDocumentId() == null){
      return new WrapperGetDocumentsResponse();
    }
    documentId = documentResponse.getDocumentId();
    Optional<FileEntity> fileEntity = repository.getByKey(documentId,null);

    if(fileEntity.isEmpty()){
      return new WrapperGetDocumentsResponse();
    }
    url = generateDownloadUrl(fileEntity.get().getUrl());

    return mapper.donwloadMapper(url);
  }

  /**
   * Metodo de salida para validar otp
   *
   * @param criteria valor del onjecto request
   * @return WrapperClassifyEmail objecto de response valid
   */
  @Override
  public WrapperPostSearchDocumentsResponse searchDocuments(
          WrapperPostSearchDocumentsRequest criteria) {
    var paramters = criteria.getSearchParameters();
    if (paramters == null){
      return new WrapperPostSearchDocumentsResponse();
    }
    if(paramters.getSearchProperties() == null || paramters.getSearchProperties().isEmpty()){
      return new WrapperPostSearchDocumentsResponse();
    }
    if(paramters.getDiscriminator() == null){
      return new WrapperPostSearchDocumentsResponse();
    }

    FileSearch search = FileSearchMapper.toSearch(paramters);
    var objectListing = repository.search(search);
    return mapper.searchMapper(objectListing);
  }

  /**
   * Metodo de salida para validar otp
   *
   * @param criteria valor del onjecto request
   * @return WrapperClassifyEmail objecto de response valid
   */
  @Override
  public WrapperPostDocumentsResponse uploadDocument(WrapperPostDocumentsRequest criteria) {



    String documentId = UUID.randomUUID().toString();


    try {
      var document = criteria.getDocument();
      if(document == null){
        return new WrapperPostDocumentsResponse();
      }
      if (document.getFolderReference()==null){
        return new WrapperPostDocumentsResponse();
      }
      if (document.getAclReference()==null){
        return new WrapperPostDocumentsResponse();
      }
      byte[] fileBytes = Base64.getDecoder().decode(document.getFolderReference());
      String typeData = "";
      if("biometric".equals(criteria.getCallerInformation().getAppId())){
        typeData = getMimeTypeOfBase64(fileBytes);
      }
      String userFolder = validateRequestAndGetFolderReference(criteria,documentId,typeData);
      String key = PREFIJO_KEY+userFolder;
              //generateKey(documentId, userFolder);
      PutObjectRequest request = PutObjectRequest.builder()
              .bucket(bucketName)
              .key(key)
              .contentType(document.getMimeType())
              .build();

      s3Client.putObject(request, RequestBody.fromBytes(fileBytes));

      // ✅ mapear entity
      FileEntity entity = mapper.toEntity(criteria, documentId, key);

      // guardar
      entity = repository.putIfAbsent(entity);

      return mapper.toDTOUpload(entity);

    } catch (Exception ex) {
      log.error("Error uploading document", ex);
      log.error(String.valueOf(ex));
      throw new RuntimeException("Error uploading document to S3", ex);
    }
  }

  private String validateRequestAndGetFolderReference(WrapperPostDocumentsRequest criteria, String id, String typeData) {

    if (criteria == null || criteria.getDocument() == null) {
      throw new IllegalArgumentException("Document is required");
    }

    var doc = criteria.getDocument();
    if (doc == null){
      throw new IllegalArgumentException("File (base64) is required");
    }

    List<WrapperMySearchDocumentResponseOwnersInner> name = doc.getOwners();
    if (name == null || name.isEmpty()) {
      throw new IllegalArgumentException("File owner is required");
    }
    String dni = name.get(0).getOwnerId();
    if (dni== null || dni.isEmpty()) {
      throw new IllegalArgumentException("File owner is required");
    }
    if(typeData == null || typeData.isEmpty()){
      return dni+"/"+doc.getName();
    }
    return dni+"/"+id+typeData;
  }


  /**
   * Metodo de salida para validar otp
   *
   * @param documentId valor del onjecto request
   * @return WrapperClassifyEmail objecto de response valid
   */
  @Override
  public WrapperMySearchDocumentResponse getDocument(String documentId) {
    var objectMetadataOptional =repository.getByKey(documentId,null);
    if(objectMetadataOptional.isEmpty()){
      return new WrapperMySearchDocumentResponse();
    }
    var objectMetadata = objectMetadataOptional.get();
    return mapper.getMapper(objectMetadata);
  }

  @Override
  public List<WrapperMySearchDocumentVersionResponse> getDocumentVersion(String id) {
    var objectMetadataOptional =repository.findAll();
    return mapper.getVersions(objectMetadataOptional);
  }

  /**
   * Metodo de salida para validar otp
   *
   * @param documentId valor del onjecto request
   * @return WrapperClassifyEmail objecto de response valid
   */
  @Override
  public Void deleteDocument(String documentId) {
    Optional<FileEntity> fileEntityOptional = repository.getByKey(documentId,null);
    if (fileEntityOptional.isEmpty()){
      return null;
    }
    FileEntity fileEntity = fileEntityOptional.get();

    String key = fileEntity.getUrl();
    DeleteObjectRequest request = DeleteObjectRequest.builder()
            .bucket(bucketName)
            .key(key)
            .build();

    s3Client.deleteObject(request);
    repository.delete(fileEntity);
    return null;
  }

  private String generateDownloadUrl(String key) {

    GetObjectRequest getObjectRequest = GetObjectRequest.builder()
            .bucket(bucketName)
            .key(key)
            .build();

    GetObjectPresignRequest presignRequest =
            GetObjectPresignRequest.builder()
                    .signatureDuration(Duration.ofMinutes(10)) // expira en 10 min
                    .getObjectRequest(getObjectRequest)
                    .build();

    PresignedGetObjectRequest presignedRequest =
            s3Presigner.presignGetObject(presignRequest);

    return presignedRequest.url().toString();
  }

}
