
package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output;

import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.input.rest.data.*;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.jpa.FileEntity;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.jpa.FileEntityRepository;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils.FileSearch;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.*;

import org.mockito.junit.jupiter.MockitoExtension;
import software.amazon.awssdk.core.sync.RequestBody;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.PutObjectRequest;
import software.amazon.awssdk.services.s3.presigner.S3Presigner;

import java.util.*;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
class DocumentManagementAdapterTest {

    @Mock
    private S3Client s3Client;

    @Mock
    private S3Presigner s3Presigner;

    @Mock
    private DocumentMapper mapper;

    @Mock
    private FileEntityRepository repository;

    @InjectMocks
    private DocumentManagementAdapter adapter;

    @BeforeEach
    void setup() {
        adapter.bucketName = "test-bucket";
    }

    // =========================
    // downloadDocument
    // =========================

    @Test
    void downloadDocument_ok() {
        WrapperGetDocumentsRequest request = new WrapperGetDocumentsRequest();
        WrapperMyDocumentResponse doc = new WrapperMyDocumentResponse();
        doc.setDocumentId("123");
        request.setDocument(doc);

        FileEntity entity = new FileEntity();
        entity.setId("123");

        when(repository.getByKey("123", null))
                .thenReturn(Optional.of(entity));

        when(mapper.donwloadMapper(any()))
                .thenReturn(new WrapperGetDocumentsResponse());

        var response = adapter.downloadDocument(request);

        assertNotNull(response);
        verify(repository).getByKey("123", null);
    }

    @Test
    void downloadDocument_nullDocument() {
        WrapperGetDocumentsRequest request = new WrapperGetDocumentsRequest();

        var response = adapter.downloadDocument(request);

        assertNotNull(response);
    }

    @Test
    void downloadDocument_notFound() {
        WrapperGetDocumentsRequest request = new WrapperGetDocumentsRequest();
        WrapperMyDocumentResponse doc = new WrapperMyDocumentResponse();
        doc.setDocumentId("123");
        request.setDocument(doc);

        when(repository.getByKey("123", null))
                .thenReturn(Optional.empty());

        var response = adapter.downloadDocument(request);

        assertNotNull(response);
    }

    // =========================
    // searchDocuments
    // =========================

    @Test
    void searchDocuments_ok() {
        WrapperPostSearchDocumentsRequest request = new WrapperPostSearchDocumentsRequest();

        SearchParameters params = new SearchParameters();
        params.setDiscriminator("AND");

        SearchProperty prop = new SearchProperty();
        prop.setKeyId("NAME");
        prop.setKeyOperator("Equal");
        prop.setKeyValue("test");

        params.setSearchProperties(List.of(prop));
        request.setSearchParameters(params);

        when(repository.search(any(FileSearch.class)))
                .thenReturn(List.of(new FileEntity()));

        when(mapper.searchMapper(any()))
                .thenReturn(new WrapperPostSearchDocumentsResponse());

        var response = adapter.searchDocuments(request);

        assertNotNull(response);
        verify(repository).search(any());
    }

    @Test
    void searchDocuments_invalidParams() {
        WrapperPostSearchDocumentsRequest request = new WrapperPostSearchDocumentsRequest();

        var response = adapter.searchDocuments(request);

        assertNotNull(response);
    }

    // =========================
    // uploadDocument
    // =========================

    @Test
    void uploadDocument_ok() {
        WrapperPostDocumentsRequest request = new WrapperPostDocumentsRequest();
        WrapperMyDocumentRequest doc = new WrapperMyDocumentRequest();

        doc.setAclReference("user1");
        doc.setFolderReference(Base64.getEncoder().encodeToString("file".getBytes()));
        doc.setMimeType("application/pdf");
        doc.setName("file.pdf");

        request.setDocument(doc);

        FileEntity entity = new FileEntity();
        entity.setId("123");

        when(mapper.toEntity(any())).thenReturn(entity);
        when(repository.putIfAbsent(any())).thenReturn(entity);
        when(mapper.toDTOUpload(any())).thenReturn(new WrapperPostDocumentsResponse());

        var response = adapter.uploadDocument(request);

        assertNotNull(response);
        verify(s3Client).putObject(any(PutObjectRequest.class), any(RequestBody.class));
    }

    @Test
    void uploadDocument_invalidRequest() {
        WrapperPostDocumentsRequest request = new WrapperPostDocumentsRequest();

        assertThrows(IllegalArgumentException.class,
                () -> adapter.uploadDocument(request));
    }

    // =========================
    // getDocument
    // =========================

    @Test
    void getDocument_ok() {
        FileEntity entity = new FileEntity();
        entity.setId("123");

        when(repository.getByKey("123", null))
                .thenReturn(Optional.of(entity));

        when(mapper.getMapper(any()))
                .thenReturn(new WrapperMySearchDocumentResponse());

        var response = adapter.getDocument("123");

        assertNotNull(response);
    }

    @Test
    void getDocument_notFound() {
        when(repository.getByKey("123", null))
                .thenReturn(Optional.empty());

        var response = adapter.getDocument("123");

        assertNotNull(response);
    }

    // =========================
    // deleteDocument
    // =========================

    @Test
    void deleteDocument_ok() {
        FileEntity entity = new FileEntity();
        entity.setId("123");
        entity.setName("file.pdf");
        entity.setCustomerId("user1");

        when(repository.getByKey("123", null))
                .thenReturn(Optional.of(entity));

        adapter.deleteDocument("123");

        verify(s3Client).deleteObject(any());
        verify(repository).delete(entity);
    }

    @Test
    void deleteDocument_notFound() {
        when(repository.getByKey("123", null))
                .thenReturn(Optional.empty());

        var result = adapter.deleteDocument("123");

        assertNull(result);
    }
}
