@ExtendWith(MockitoExtension.class)
class TokenServiceTest {

    @Mock
    private WebClient.Builder webClientBuilder;

    @Mock
    private WebClient webClient;

    @Mock
    private WebClient.RequestBodyUriSpec requestBodyUriSpec;

    @Mock
    private WebClient.RequestHeadersSpec<?> requestHeadersSpec;

    @Mock
    private WebClient.ResponseSpec responseSpec;

    private TokenService tokenService;

    @BeforeEach
    void setUp() {
        when(webClientBuilder.baseUrl(anyString())).thenReturn(webClientBuilder);
        when(webClientBuilder.build()).thenReturn(webClient);

        tokenService = new TokenService(webClientBuilder);
    }

    @Test
    void shouldGetTokenSuccessfully() {

        TokenResponse tokenResponse = new TokenResponse();
        tokenResponse.setAccessToken("token-ok");

        when(webClient.post()).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.uri(anyString())).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.contentType(any())).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.body(any())).thenReturn(requestHeadersSpec);
        when(requestHeadersSpec.retrieve()).thenReturn(responseSpec);

        when(responseSpec.onStatus(any(), any())).thenReturn(responseSpec);

        when(responseSpec.bodyToMono(TokenResponse.class))
                .thenReturn(Mono.just(tokenResponse));

        StepVerifier.create(tokenService.getToken())
                .expectNext("token-ok")
                .verifyComplete();
    }

    @Test
    void shouldHandleErrorFromApi() {

        when(webClient.post()).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.uri(anyString())).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.contentType(any())).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.body(any())).thenReturn(requestHeadersSpec);
        when(requestHeadersSpec.retrieve()).thenReturn(responseSpec);

        when(responseSpec.onStatus(any(), any())).thenReturn(responseSpec);

        when(responseSpec.bodyToMono(TokenResponse.class))
                .thenReturn(Mono.error(new RuntimeException("fail")));

        StepVerifier.create(tokenService.getToken())
                .expectError(RuntimeException.class)
                .verify();
    }
}

@Test
void shouldTriggerOnStatusErrorHandler() {

    when(webClient.post()).thenReturn(requestBodyUriSpec);
    when(requestBodyUriSpec.uri(anyString())).thenReturn(requestBodyUriSpec);
    when(requestBodyUriSpec.contentType(any())).thenReturn(requestBodyUriSpec);
    when(requestBodyUriSpec.body(any())).thenReturn(requestHeadersSpec);
    when(requestHeadersSpec.retrieve()).thenReturn(responseSpec);

    when(responseSpec.onStatus(any(), any())).thenAnswer(invocation -> {
        var function = invocation.getArgument(1);

        // simulamos error response
        var clientResponse = mock(org.springframework.web.reactive.function.client.ClientResponse.class);
        when(clientResponse.bodyToMono(String.class))
                .thenReturn(Mono.just("error-body"));

        return ((Mono<?>) function.apply(clientResponse)).thenReturn(responseSpec);
    });

    when(responseSpec.bodyToMono(TokenResponse.class))
            .thenReturn(Mono.empty());

    StepVerifier.create(tokenService.getToken())
            .verifyComplete();
}

@Test
void shouldAddAuthorizationHeaderInFilter() {

    TokenService tokenService = mock(TokenService.class);
    when(tokenService.getToken()).thenReturn(Mono.just("fake-token"));

    XmlWebClientConfig config = new XmlWebClientConfig();

    WebClient client = config.xmlWebClient(tokenService);

    // Creamos request mock
    var request = org.springframework.web.reactive.function.client.ClientRequest
            .create(org.springframework.http.HttpMethod.GET, java.net.URI.create("http://test"))
            .build();

    var exchangeFunction = mock(org.springframework.web.reactive.function.client.ExchangeFunction.class);

    when(exchangeFunction.exchange(any()))
            .thenReturn(Mono.just(mock(org.springframework.web.reactive.function.client.ClientResponse.class)));

    // Ejecutamos filtro manualmente
    var filter = config.getClass()
            .getDeclaredMethods()[1]; // hack simple (addTokenFilter)

    filter.setAccessible(true);

    try {
        var exchangeFilterFunction = (org.springframework.web.reactive.function.client.ExchangeFilterFunction)
                filter.invoke(config, tokenService);

        StepVerifier.create(exchangeFilterFunction.filter(request, exchangeFunction))
                .expectNextCount(1)
                .verifyComplete();

    } catch (Exception e) {
        throw new RuntimeException(e);
    }
}
