package com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa;


import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.application.ports.output.KnowYourCustomerRepositoryPort;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionRequest;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionResponse;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.data.Questionnairy;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.mapper.KYCEntityMapper;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.repository.KnokYourCustomerRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.util.List;

@Component
@RequiredArgsConstructor
public class KnowYourCustomerAdapter implements KnowYourCustomerRepositoryPort {

  private final KnokYourCustomerRepository repository;
  private final KYCEntityMapper mapper;

  @Override
  public WrapperPostKycResolutionResponse createProcesstKyc(
      WrapperPostKycResolutionRequest criteria) {
    List<Questionnairy> questionnairies = repository.saveAll(
            mapper.WrapperPostKycResolutionRequestToCustomerEntitys(
                    criteria.getKnowYourCustomerQuestionnaires()));
    String id = questionnairies.get(0).getDni();
    return mapper.toDto(
            questionnairies, id
            );


  }
}



package com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.mapper;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.json.JsonMapper;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionRequest;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionRequestKnowYourCustomerQuestionnairesInner;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionResponse;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.data.Questionnairy;
import org.mapstruct.Mapper;
import org.mapstruct.Mapping;

import java.util.List;

@Mapper(componentModel = "spring")
public interface KYCEntityMapper {

    @Mapping(target = "id", ignore = true)
    @Mapping(target = "dni", source = "questionnaireId")
    @Mapping(target = "audit", expression = "java(toStringJson(request.getAudit())")
    @Mapping(target = "questions", expression = "java(toStringJson(request.getQuestions())")
    @Mapping(target = "version", defaultValue = "1")
    Questionnairy WrapperPostKycResolutionRequestToCustomerEntity(WrapperPostKycResolutionRequestKnowYourCustomerQuestionnairesInner request);

    List<Questionnairy> WrapperPostKycResolutionRequestToCustomerEntitys(List<WrapperPostKycResolutionRequestKnowYourCustomerQuestionnairesInner> request);

    @Mapping(target = "kycResolutionId", source = "id")
    WrapperPostKycResolutionResponse toDto(List<Questionnairy> questionnairies, String id);

    default String toStringJson(Object o){
        JsonMapper jsonMapper = new JsonMapper();
        try {
            return jsonMapper.writeValueAsString(o);
        } catch (JsonProcessingException e) {
            return "Error al parsear objeto";
        }
    }
}
