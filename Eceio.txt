package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.util;

import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Base64;

import static com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.constants.Constant.PREFIJO_KEY;
import static com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.constants.Constant.TYPE_PDF;

public final class UtilsFile {
    private UtilsFile() {
        // Evita instanciación
    }

    public static String generateFileName(String name){
        return name+TYPE_PDF;
    }

    public static String generateKey(String id){
        return PREFIJO_KEY+generateFileName(id);
    }

    public static String getDocumentId(String key){
        return getFileName(key).replace(TYPE_PDF,"");
    }

    public static String getFileName(String key){
        return key.replace(PREFIJO_KEY,"");
    }

    public static File base64ToFile(String base64, String name) throws IOException {
        if (base64 == null){
            return null;
        }
        byte[] decodedBytes = Base64.getDecoder().decode(base64);
        Path destinationFile = Paths.get("./"+name);
        return Files.write(destinationFile, decodedBytes).toFile();
    }
}





package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output;

import com.amazonaws.services.s3.model.ObjectListing;
import com.amazonaws.services.s3.model.ObjectMetadata;
import com.amazonaws.services.s3.model.S3ObjectSummary;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.input.rest.data.*;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.io.File;
import java.util.List;
import java.util.UUID;
import java.util.stream.Collectors;

import static com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.util.UtilsFile.*;

@Component
@Slf4j
public class DocumentMapper {

    public WrapperGetDocumentsResponse donwloadMapper(String url){
        WrapperGetDocumentsResponse response = new WrapperGetDocumentsResponse();
        LinkReference linkReference = new LinkReference();
        linkReference.setHref(url);
        response.setAccessInformation(linkReference);
        return response;
    }

    public WrapperPostSearchDocumentsResponse searchMapper(ObjectListing objectListing){
        WrapperPostSearchDocumentsResponse response = new WrapperPostSearchDocumentsResponse();
        List<WrapperMySearchDocumentResponse> documents = objectListing.getObjectSummaries().stream().map(objectSumary ->{
            WrapperMySearchDocumentResponse wrapper = new WrapperMySearchDocumentResponse();
            wrapper.setDocumentId(getDocumentId(objectSumary.getKey()));
            wrapper.setName(getFileName(objectSumary.getKey()));
            return wrapper;
        }).collect(Collectors.toList());
        response.setDocuments(documents);
        return response;
    }

    public WrapperPostDocumentsResponse uploadMapper(String documentId){
        WrapperMyDocumentResponse documentResponse = new WrapperMyDocumentResponse();
        documentResponse.documentId(documentId);
        WrapperPostDocumentsResponse response = new WrapperPostDocumentsResponse();
        response.setDocument(documentResponse);
        return response;
    }

    public WrapperMySearchDocumentResponse getMapper(String documentId, ObjectMetadata objectMetadata){
        WrapperMySearchDocumentResponse response = new WrapperMySearchDocumentResponse();
        response.setName(generateFileName(documentId));
        response.setDocumentId(documentId);
        response.setStatus((new WrapperMySearchDocumentResponseStatus()).statusCode(objectMetadata.getArchiveStatus()));
        response.setMimeType(objectMetadata.getContentType());
        return response;
    }

}





package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output;


import com.amazonaws.services.s3.AmazonS3;
import com.amazonaws.services.s3.model.ObjectListing;
import com.amazonaws.services.s3.model.ObjectMetadata;
import com.amazonaws.services.s3.model.S3Object;
import com.amazonaws.services.s3.model.S3ObjectSummary;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.application.ports.output.DocumentManagementRepositoryPort;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.input.rest.data.*;
import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.constants.Constant;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.io.File;
import java.util.List;
import java.util.UUID;
import java.util.stream.Collectors;

import static com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.util.UtilsFile.*;


@Component
@RequiredArgsConstructor
@Slf4j
public class DocumentManagementAdapter implements DocumentManagementRepositoryPort {

  private final AmazonS3 amazonS3;

  private final DocumentMapper mapper;

  @Value("${aws.bucket}")
  private String bucketName;

  @Override
  public WrapperGetDocumentsResponse downloadDocument(WrapperGetDocumentsRequest criteria) {
    String fileName = generateKey(criteria.getDocument().getDocumentId());
    String url =  amazonS3.getUrl(bucketName,fileName).getRef();
    return mapper.donwloadMapper(url);
  }

  @Override
  public WrapperPostSearchDocumentsResponse searchDocuments(
      WrapperPostSearchDocumentsRequest criteria) {
    ObjectListing objectListing = amazonS3.listObjects(bucketName);
    return mapper.searchMapper(objectListing);
  }

  @Override
  public WrapperPostDocumentsResponse uploadDocument(WrapperPostDocumentsRequest criteria) {
    String documentId = UUID.randomUUID().toString();
    try {
        assert criteria.getDocument() != null;
        File file = base64ToFile(criteria.getDocument().getFolderReference(),generateFileName(documentId));
      amazonS3.putObject(bucketName,generateKey(documentId),file);
      return mapper.uploadMapper(documentId);
    } catch (Exception ex){
      log.error("error : "+ex);
    }
    return new WrapperPostDocumentsResponse();
  }

  @Override
  public WrapperMySearchDocumentResponse getDocument(String documentId) {
    ObjectMetadata objectMetadata =amazonS3.getObjectMetadata(bucketName,generateKey(documentId));
    return mapper.getMapper(documentId,objectMetadata);
  }

  @Override
  public Void deleteDocument(String documentId) {
    String filename = generateKey(documentId);
    amazonS3.deleteObject(bucketName,filename);
    return null;
  }


}
