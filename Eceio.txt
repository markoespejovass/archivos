import lombok.RequiredArgsConstructor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import software.amazon.awssdk.http.apache.ApacheHttpClient;
import software.amazon.awssdk.http.apache.ProxyConfiguration;
import software.amazon.awssdk.http.SdkHttpClient;

@Configuration
@RequiredArgsConstructor
public class AwsHttpClientConfig {

    private final ProxyProperties proxyProperties;

    @Bean
    public SdkHttpClient sdkHttpClient() {

        return ApacheHttpClient.builder()
                .proxyConfiguration(
                        ProxyConfiguration.builder()
                                .endpoint(
                                        URI.create(
                                                "http://" +
                                                        proxyProperties.getHost() +
                                                        ":" +
                                                        proxyProperties.getPort()
                                        )
                                )
                                .build()
                )
                .build();
    }
}


@Bean
public StsClient stsClient(SdkHttpClient sdkHttpClient) {

    return StsClient.builder()
            .region(Region.of(awsRegion))
            .httpClient(sdkHttpClient)
            .credentialsProvider(DefaultCredentialsProvider.create())
            .build();
}


@Bean
public S3Client s3Client(
        StsClient stsClient,
        SdkHttpClient sdkHttpClient) {

    StsAssumeRoleCredentialsProvider credentialsProvider =
            StsAssumeRoleCredentialsProvider.builder()
                    .stsClient(stsClient)
                    .refreshRequest(AssumeRoleRequest.builder()
                            .roleArn(dynamoRoleArn)
                            .roleSessionName("s3-cross-account-session")
                            .build())
                    .build();

    return S3Client.builder()
            .region(Region.of(awsRegion))
            .httpClient(sdkHttpClient)
            .credentialsProvider(credentialsProvider)
            .build();
}
