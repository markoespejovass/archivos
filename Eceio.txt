package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils;

import org.junit.jupiter.api.Test;

import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

class ConditionTest {

    @Test
    void constructor_withOperator() {
        Condition cond = new Condition("name", Operator.EQ, "test");

        assertEquals("name", cond.field);
        assertEquals(Operator.EQ, cond.operator);
        assertEquals("test", cond.value);
        assertNull(cond.values);
    }

    @Test
    void constructor_withInOperator() {
        Condition cond = new Condition("status", List.of("A", "B"));

        assertEquals("status", cond.field);
        assertEquals(Operator.IN, cond.operator);
        assertNotNull(cond.values);
        assertEquals(2, cond.values.size());
    }
}

package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils;

import org.junit.jupiter.api.Test;

import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

class FileSearchTest {

    @Test
    void builder_andCondition() {
        FileSearch search = FileSearch.builder()
                .and("name", Operator.EQ, "test")
                .build();

        assertFalse(search.useOr);
        assertEquals(1, search.conditions.size());
    }

    @Test
    void builder_orCondition() {
        FileSearch search = FileSearch.builder()
                .or("name", Operator.EQ, "test")
                .build();

        assertTrue(search.useOr);
        assertEquals(1, search.conditions.size());
    }

    @Test
    void builder_inCondition() {
        FileSearch search = FileSearch.builder()
                .in("status", List.of("A", "B"))
                .build();

        assertEquals(1, search.conditions.size());
    }

    @Test
    void builder_groupCondition() {
        FileSearch group = FileSearch.builder()
                .and("status", Operator.EQ, "A")
                .build();

        FileSearch search = FileSearch.builder()
                .andGroup(group)
                .build();

        assertEquals(1, search.conditions.size());
    }
}


package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils;

import org.junit.jupiter.api.Test;
import software.amazon.awssdk.enhanced.dynamodb.Expression;

import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

class ExpressionBuilderTest {

    @Test
    void shouldBuildSimpleExpression() {
        FileSearch search = FileSearch.builder()
                .and("name", Operator.EQ, "test")
                .build();

        ExpressionBuilder builder = new ExpressionBuilder();
        Expression expr = builder.build(search);

        assertNotNull(expr);
        assertTrue(expr.expression().contains("="));
    }

    @Test
    void shouldBuildContainsExpression() {
        FileSearch search = FileSearch.builder()
                .and("name", Operator.CONTAINS, "pdf")
                .build();

        Expression expr = new ExpressionBuilder().build(search);

        assertTrue(expr.expression().contains("contains"));
    }

    @Test
    void shouldBuildInExpression() {
        FileSearch search = FileSearch.builder()
                .in("status", List.of("A", "B"))
                .build();

        Expression expr = new ExpressionBuilder().build(search);

        assertTrue(expr.expression().contains("IN"));
    }

    @Test
    void shouldBuildOrExpression() {
        FileSearch search = FileSearch.builder()
                .or("name", Operator.EQ, "A")
                .or("status", Operator.EQ, "B")
                .build();

        Expression expr = new ExpressionBuilder().build(search);

        assertTrue(expr.expression().contains("OR"));
    }

    @Test
    void shouldThrowForUnsupportedType() {
        FileSearch search = FileSearch.builder()
                .and("name", Operator.EQ, new Object()) // tipo inválido
                .build();

        ExpressionBuilder builder = new ExpressionBuilder();

        assertThrows(IllegalArgumentException.class, () -> builder.build(search));
    }
}




package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.config;

import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.utils.FileSearch;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.*;
import software.amazon.awssdk.enhanced.dynamodb.*;

import java.util.List;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

class BaseDynamoRepositoryTest {

    @Mock
    DynamoDbEnhancedClient enhancedClient;

    @Mock
    DynamoDbTable<String> table;

    BaseDynamoRepository<String> repository;

    @BeforeEach
    void setup() {
        MockitoAnnotations.openMocks(this);

        when(enhancedClient.table(anyString(), any()))
                .thenReturn(table);

        repository = new BaseDynamoRepository<>(
                enhancedClient,
                "test-table",
                TableSchema.fromBean(String.class),
                String.class,
                "id",
                null
        ) {};
    }

    @Test
    void putIfAbsent_shouldCallDynamo() {
        repository.putIfAbsent("test");

        verify(table).putItem(any(PutItemEnhancedRequest.class));
    }

    @Test
    void delete_shouldCallDynamo() {
        repository.delete("test");

        verify(table).deleteItem("test");
    }

    @Test
    void search_shouldCallScan() {
        FileSearch search = FileSearch.builder()
                .and("name", Operator.EQ, "test")
                .build();

        PageIterable<String> iterable = mock(PageIterable.class);
        when(table.scan(any(ScanEnhancedRequest.class)))
                .thenReturn(iterable);

        when(iterable.items()).thenReturn(List.of("a", "b"));

        List<String> result = repository.search(search);

        assertEquals(2, result.size());
    }
}
