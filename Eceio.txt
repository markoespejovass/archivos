@ExtendWith(MockitoExtension.class)
class DocumentManagementAdapterTest {

    @Mock
    private AmazonS3 amazonS3;

    @Mock
    private DocumentMapper mapper;

    @Mock
    private FileEntityRepository repository;

    @InjectMocks
    private DocumentManagementAdapter adapter;

    private final String bucket = "test-bucket";

    @BeforeEach
    void setup() {
        ReflectionTestUtils.setField(adapter, "bucketName", bucket);
    }

    // =========================
    // DOWNLOAD
    // =========================
    @Test
    void shouldDownloadDocument() {
        WrapperGetDocumentsRequest request = new WrapperGetDocumentsRequest();
        var doc = new WrapperGetDocumentsRequest.Document();
        doc.setDocumentId("123");
        request.setDocument(doc);

        FileEntity entity = new FileEntity();
        entity.setUrl("http://url");

        WrapperGetDocumentsResponse response = new WrapperGetDocumentsResponse();

        when(repository.findById("123")).thenReturn(Optional.of(entity));
        when(mapper.donwloadMapper("http://url")).thenReturn(response);

        WrapperGetDocumentsResponse result = adapter.downloadDocument(request);

        assertEquals(response, result);
    }

    // =========================
    // SEARCH
    // =========================
    @Test
    void shouldSearchDocuments() {
        WrapperPostSearchDocumentsRequest request = new WrapperPostSearchDocumentsRequest();
        var searchParams = new WrapperPostSearchDocumentsRequest.SearchParameters();
        searchParams.setSearchProperties(List.of());
        searchParams.setDiscriminator("test");
        request.setSearchParameters(searchParams);

        List<FileEntity> entities = List.of(new FileEntity());
        WrapperPostSearchDocumentsResponse response = new WrapperPostSearchDocumentsResponse();

        when(repository.findAll(any(Specification.class))).thenReturn(entities);
        when(mapper.searchMapper(entities)).thenReturn(response);

        WrapperPostSearchDocumentsResponse result = adapter.searchDocuments(request);

        assertEquals(response, result);
    }

    // =========================
    // UPLOAD SUCCESS
    // =========================
    @Test
    void shouldUploadDocumentSuccessfully() throws Exception {
        WrapperPostDocumentsRequest request = new WrapperPostDocumentsRequest();
        var doc = new WrapperPostDocumentsRequest.Document();

        String base64 = Base64.getEncoder().encodeToString("file".getBytes());

        doc.setFolderReference(base64);
        doc.setName("file.pdf");
        doc.setMimeType("application/pdf");

        request.setDocument(doc);

        FileEntity entity = new FileEntity();
        WrapperPostDocumentsResponse response = new WrapperPostDocumentsResponse();

        when(mapper.toEntity(request)).thenReturn(entity);
        when(repository.save(any())).thenReturn(entity);
        when(mapper.toDTOUpload(entity)).thenReturn(response);
        when(amazonS3.getUrl(any(), any())).thenReturn(new URL("http://s3-url"));

        WrapperPostDocumentsResponse result = adapter.uploadDocument(request);

        assertNotNull(result);
        verify(amazonS3).putObject(eq(bucket), anyString(), any(ByteArrayInputStream.class), any(ObjectMetadata.class));
        verify(repository).save(any());
    }

    // =========================
    // UPLOAD VALIDATIONS
    // =========================
    @Test
    void shouldThrowWhenRequestIsNull() {
        assertThrows(IllegalArgumentException.class, () -> adapter.uploadDocument(null));
    }

    @Test
    void shouldThrowWhenDocumentIsNull() {
        WrapperPostDocumentsRequest request = new WrapperPostDocumentsRequest();
        assertThrows(IllegalArgumentException.class, () -> adapter.uploadDocument(request));
    }

    @Test
    void shouldThrowWhenBase64IsEmpty() {
        WrapperPostDocumentsRequest request = new WrapperPostDocumentsRequest();
        var doc = new WrapperPostDocumentsRequest.Document();
        doc.setFolderReference("");
        doc.setName("file.pdf");
        request.setDocument(doc);

        assertThrows(IllegalArgumentException.class, () -> adapter.uploadDocument(request));
    }

    @Test
    void shouldThrowWhenNameIsEmpty() {
        WrapperPostDocumentsRequest request = new WrapperPostDocumentsRequest();
        var doc = new WrapperPostDocumentsRequest.Document();
        doc.setFolderReference("abc");
        doc.setName("");
        request.setDocument(doc);

        assertThrows(IllegalArgumentException.class, () -> adapter.uploadDocument(request));
    }

    // =========================
    // UPLOAD ERROR
    // =========================
    @Test
    void shouldThrowRuntimeExceptionWhenS3Fails() throws Exception {
        WrapperPostDocumentsRequest request = new WrapperPostDocumentsRequest();
        var doc = new WrapperPostDocumentsRequest.Document();

        String base64 = Base64.getEncoder().encodeToString("file".getBytes());

        doc.setFolderReference(base64);
        doc.setName("file.pdf");
        doc.setMimeType("application/pdf");
        request.setDocument(doc);

        when(amazonS3.getUrl(any(), any())).thenThrow(new RuntimeException("S3 error"));

        assertThrows(RuntimeException.class, () -> adapter.uploadDocument(request));
    }

    // =========================
    // GET DOCUMENT
    // =========================
    @Test
    void shouldGetDocument() {
        FileEntity entity = new FileEntity();
        WrapperMySearchDocumentResponse response = new WrapperMySearchDocumentResponse();

        when(repository.findById("123")).thenReturn(Optional.of(entity));
        when(mapper.getMapper(entity)).thenReturn(response);

        WrapperMySearchDocumentResponse result = adapter.getDocument("123");

        assertEquals(response, result);
    }

    // =========================
    // DELETE DOCUMENT
    // =========================
    @Test
    void shouldDeleteDocument() {
        FileEntity entity = new FileEntity();
        entity.setId("123");
        entity.setName("file");

        when(repository.findById("123")).thenReturn(Optional.of(entity));

        adapter.deleteDocument("123");

        verify(amazonS3).deleteObject(eq(bucket), anyString());
        verify(repository).delete(entity);
    }
}
