@Bean
public S3Client s3Client() {
    return S3Client.builder()
            .endpointOverride(
                URI.create("http://localhost:4566"))
            .region(Region.US_EAST_1)
            .credentialsProvider(
                StaticCredentialsProvider.create(
                    AwsBasicCredentials.create("test", "test")
                ))
            .forcePathStyle(true)
            .build();
}
