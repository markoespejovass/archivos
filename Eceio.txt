var response =
    webClient
        .post()
        .uri(Constant.POST_CIVIL_VALIDATION)
        .bodyValue(request)
        .retrieve()
        .onStatus(
            HttpStatusCode::isError,
            clientResponse ->
                clientResponse.bodyToMono(String.class)
                    .flatMap(errorBody -> {
                        log.error("Facephi error body: {}", errorBody);
                        return Mono.error(
                            new RuntimeException(errorBody));
                    })
        )
        .bodyToMono(FacephiCivilValidationResponse.class)
        .block();
