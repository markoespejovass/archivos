package com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.domain;

import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.application.ports.output.KnowYourCustomerRepositoryPort;
import lombok.AllArgsConstructor;

import java.util.List;
import java.util.Map;

@AllArgsConstructor
public class ScoreCalculator {

    private final KnowYourCustomerRepositoryPort repositoryPort;

    public double calculateScore(
            String factorCode, Map<String,String> kycData
    ){
        List<WeightRule> rules = repositoryPort.getRulesByFactor(factorCode);

       return rules.stream().mapToDouble(rule ->{

           String inputCode =  rule.getCode();
           double value;

           if(isBaseField(inputCode)){
               String textValue = kycData.get(inputCode);
               value = repositoryPort.getValue(inputCode,textValue);
           } else {
               value = calculateScore(inputCode,kycData);
           }
           return value * rule.getWeight();
       }).sum();
    }

    private boolean isBaseField(String inputCode){
        return inputCode.length()<=3;
    }

}




package com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa;


import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.application.ports.output.KnowYourCustomerRepositoryPort;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.domain.WeightRule;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionRequest;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionResponse;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.data.Kyc;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.data.KycEntity;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.mapper.KYCEntityMapper;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.mapper.WeightMapper;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.repository.KnokYourCustomerRepository;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.repository.ValueRepository;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.repository.WeightRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.util.List;

@Component
@RequiredArgsConstructor
public class KnowYourCustomerAdapter implements KnowYourCustomerRepositoryPort {

  private final KnokYourCustomerRepository repository;
  private final WeightRepository weightRepository;
  private final KYCEntityMapper mapper;
  private final WeightMapper mapperWeight;
  private final ValueRepository valueRepository;

  @Override
  public WrapperPostKycResolutionResponse createProcesstKyc(
      WrapperPostKycResolutionRequest criteria) {
    Kyc kyc = repository.save(
            mapper.WrapperPostKycResolutionRequestToKycEntity(criteria));
    if (kyc.getId() == 0){
      return new WrapperPostKycResolutionResponse();
    }
    String id = kyc.getDocumentNumber();
    return mapper.toDto(id);


  }

  @Override
  public Kyc getKyc(String documento){
    return repository.findByDocumentNumber(documento);
  }

  @Override
  public Double getValue(String code, String input) {
    return Double.valueOf(String.valueOf(valueRepository.findByInputCode(input).getValueNumeric()));
  }

  @Override
  public List<WeightRule> getRulesByFactor(String factor) {
    return mapperWeight.toDTOList(weightRepository.findByFactorCode(factor));
  }
}
