
@ExtendWith(MockitoExtension.class)
class KnowYourCustomerAdapterTest {

    @Mock
    private KnokYourCustomerRepository repository;
    @Mock
    private WeightRepository weightRepository;
    @Mock
    private KYCEntityMapper mapper;
    @Mock
    private WeightMapper mapperWeight;
    @Mock
    private ValueRepository valueRepository;
    @Mock
    private InputDefinitionRepository inputDefinitionRepository;

    @InjectMocks
    private KnowYourCustomerAdapter adapter;

    @Test
    void shouldCreateKycSuccessfully() {
        WrapperPostKycResolutionRequest request = new WrapperPostKycResolutionRequest();
        Kyc kyc = new Kyc();
        kyc.setId(1);
        kyc.setDocumentNumber("12345678");

        when(mapper.WrapperPostKycResolutionRequestToKycEntity(request)).thenReturn(kyc);
        when(repository.save(any())).thenReturn(kyc);
        when(mapper.toDto("12345678")).thenReturn(new WrapperPostKycResolutionResponse());

        WrapperPostKycResolutionResponse response = adapter.createProcesstKyc(request);

        assertNotNull(response);
    }

    @Test
    void shouldReturnEmptyResponseWhenIdIsZero() {
        WrapperPostKycResolutionRequest request = new WrapperPostKycResolutionRequest();
        Kyc kyc = new Kyc();
        kyc.setId(0);

        when(mapper.WrapperPostKycResolutionRequestToKycEntity(request)).thenReturn(kyc);
        when(repository.save(any())).thenReturn(kyc);

        WrapperPostKycResolutionResponse response = adapter.createProcesstKyc(request);

        assertNotNull(response);
    }

    @Test
    void shouldGetKyc() {
        Kyc kyc = new Kyc();
        when(repository.findByDocumentNumber("123")).thenReturn(kyc);

        assertEquals(kyc, adapter.getKyc("123"));
    }

    @Test
    void shouldGetValues() {
        when(valueRepository.findAll()).thenReturn(List.of(new ValueMapping()));

        assertFalse(adapter.getValues().isEmpty());
    }

    @Test
    void shouldGetInputs() {
        when(inputDefinitionRepository.findAll()).thenReturn(List.of(new InputDefinition()));

        assertFalse(adapter.getAllInputs().isEmpty());
    }

    @Test
    void shouldGetRulesByFactor() {
        when(weightRepository.findByFactorCode("F1")).thenReturn(List.of());
        when(mapperWeight.toDTOList(any())).thenReturn(List.of(new WeightRule()));

        assertFalse(adapter.getRulesByFactor("F1").isEmpty());
    }
}





class KYCEntityMapperTest {

    }

    void shouldReturnVS1() {

        assertEquals("1", mapper.getVS(request));

    @Test

        assertEquals("Régimen Reforzado", mapper.getDueDiligenceRegime(request));

    void shouldReturnDueDiligenceGeneral() {
        WrapperPostKycResolutionRequest request = TestBuilder.withPep(false, "No");

        assertEquals("Régimen General", mapper.getDueDiligenceRegime(request));

    void shouldMapEXP() {

    }

    @Test

        assertTrue(json.contains("a"));
    }
}



@ExtendWith(MockitoExtension.class)
class ScoreCalculatorTest {

    @Mock
    private KnowYourCustomerRepositoryPort repositoryPort;

    @Test
    void shouldCalculateScoreBaseField() {
        ScoreCalculator calculator = new ScoreCalculator(repositoryPort);

        WeightRule rule = new WeightRule("A", 2.0);
        when(repositoryPort.getRulesByFactor("SCORE")).thenReturn(List.of(rule));

        Map<String, String> kycData = Map.of("A", "HIGH");

        ValueMapping mapping = new ValueMapping();
        mapping.setInputCode("A");
        mapping.setValueText("HIGH");
        mapping.setValueNumeric(10);

        double result = calculator.calculateScore("SCORE", kycData, List.of(mapping));

        assertEquals(20.0, result);
    }

    @Test
    void shouldReturnZeroWhenNoMapping() {
        ScoreCalculator calculator = new ScoreCalculator(repositoryPort);

        WeightRule rule = new WeightRule("A", 1.0);
        when(repositoryPort.getRulesByFactor("SCORE")).thenReturn(List.of(rule));

        double result = calculator.calculateScore("SCORE", Map.of("A", "X"), List.of());

        assertEquals(0.0, result);
    }
}

        when(repositoryPort.getKyc(any())).thenReturn(new Kyc());

        WrapperPostRiskScoreResponse response = useCase.getRiskScore(request);

        assertNotNull(response);
        assertNotNull(response.getRisk());
    }
}
