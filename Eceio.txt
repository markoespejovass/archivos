package com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output;

import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.KnowYourCustomerAdapter;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.data.Kyc;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.*;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.mapper.KYCEntityMapper;
import com.santander.framework.springboot.core.exceptions.SantanderException;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.*;
import org.mockito.junit.jupiter.MockitoExtension;
import software.amazon.awssdk.enhanced.dynamodb.*;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
class KnowYourCustomerAdapterTest {

    @Mock
    private DynamoDbEnhancedClient dynamoDbEnhancedClient;

    @Mock
    private DynamoDbTable<Kyc> dynamoDbTable;

    @Mock
    private KYCEntityMapper mapper;

    @InjectMocks
    private KnowYourCustomerAdapter adapter;

    @BeforeEach
    void setup() {
        // inyectar nombre de tabla manualmente
        ReflectionTestUtils.setField(adapter, "tableName", "test-table");
    }

    private WrapperPostKycResolutionRequest buildRequest() {
        WrapperPostKycResolutionRequest request = mock(WrapperPostKycResolutionRequest.class);
        var party = mock(WrapperPostKycResolutionRequest.Party.class);

        when(request.getParty()).thenReturn(party);
        when(party.getPartyId()).thenReturn("123");

        return request;
    }

    @Test
    void shouldCreateKycSuccessfully() {
        var request = buildRequest();
        Kyc newEntity = new Kyc();

        when(dynamoDbEnhancedClient.table(anyString(), any()))
                .thenReturn(dynamoDbTable);

        when(dynamoDbTable.getItem(any(Key.class)))
                .thenReturn(null); // no existe → OK

        when(mapper.WrapperPostKycResolutionRequestToKycEntity(request))
                .thenReturn(newEntity);

        when(mapper.toDto(anyString()))
                .thenReturn(new WrapperPostKycResolutionResponse());

        var response = adapter.createProcesstKyc(request);

        assertNotNull(response);
        verify(dynamoDbTable).putItem(any(PutItemEnhancedRequest.class));
    }

    @Test
    void shouldThrowExceptionWhenDuplicate() {
        var request = buildRequest();

        when(dynamoDbEnhancedClient.table(anyString(), any()))
                .thenReturn(dynamoDbTable);

        when(dynamoDbTable.getItem(any(Key.class)))
                .thenReturn(new Kyc()); // ya existe

        assertThrows(SantanderException.class,
                () -> adapter.createProcesstKyc(request));
    }

    @Test
    void shouldGetKyc() {
        when(dynamoDbEnhancedClient.table(anyString(), any()))
                .thenReturn(dynamoDbTable);

        Kyc kyc = new Kyc();

        when(dynamoDbTable.getItem(any(Key.class)))
                .thenReturn(kyc);

        var result = adapter.getKyc("123");

        assertEquals(kyc, result);
    }
}





@ExtendWith(MockitoExtension.class)
class BaseDynamoRepositoryTest {

    @Mock
    DynamoDbEnhancedClient client;

    @Mock
    DynamoDbTable<Object> table;

    BaseDynamoRepository<Object> repository;

    @BeforeEach
    void setup() {
        when(client.table(anyString(), any())).thenReturn(table);

        repository = new BaseDynamoRepository<>(
                client,
                "test-table",
                TableSchema.fromBean(Object.class),
                Object.class,
                "pk",
                "sk"
        ) {};
    }

    @Test
    void shouldPutIfAbsent() {
        Object item = new Object();

        repository.putIfAbsent(item);

        verify(table).putItem(any(PutItemEnhancedRequest.class));
    }

    @Test
    void shouldGetByKey() {
        Object item = new Object();

        when(table.getItem(any())).thenReturn(item);

        var result = repository.getByKey("pk", "sk");

        assertTrue(result.isPresent());
    }

    @Test
    void shouldReturnEmptyWhenNotFound() {
        when(table.getItem(any())).thenReturn(null);

        var result = repository.getByKey("pk", "sk");

        assertTrue(result.isEmpty());
    }
}




class DynamoDBConfigTest {

    @Test
    void shouldCreateDynamoDbClient() {
        DynamoDBConfig config = new DynamoDBConfig();
        ReflectionTestUtils.setField(config, "awsRegion", "us-east-1");

        var client = config.dynamoDbClient();

        assertNotNull(client);
    }

    @Test
    void shouldCreateEnhancedClient() {
        DynamoDBConfig config = new DynamoDBConfig();

        var dynamoClient = DynamoDbClient.builder()
                .region(Region.US_EAST_1)
                .build();

        var enhanced = config.dynamoDbEnhancedClient(dynamoClient);

        assertNotNull(enhanced);
    }
}
