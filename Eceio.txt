/**
 * The class representative Otp adaptarer
 *
 * @author Create by team Proyecto [NUBE] Pe
 */
@Slf4j
@Component
@AllArgsConstructor
public class OTPServiceAdapter implements OTPServiceRepositoryPort {

    private WebClient xmlWebClient;
    private XmlMapper xmlMapper;
    private WebClient jsonWebClient;

    /**
     * Metodo de salida para validar otp
     *
     * @param otp valor del otp
     * @param criteria valor del onjecto request
     * @return WrapperClassifyEmail objecto de response valid
     */
    @Override
    public WrapperClassifyEmail validCode(String otp, WrapperRequestClassifyEmails criteria) {
        boolean isValid = UtilsOtp.esNumeroPar8DigitosValido(otp);

        if (isValid) {
            return generateResponsseValidCode(StatusInfo.StatusCodeEnum.PROCESSED);
        } else {
            return generateResponsseValidCode(StatusInfo.StatusCodeEnum.ERROR);
        }
    }

    /**
     * Metodo de salida para enviar correo
     *
     * @param criteria request
     * @return objeto de response
     */
    @Override
    public ResponseOTP sendMailing(RequestOTP criteria)  {
        log.info("request: "+ criteria.toString());

        String otp = jsonWebClient
                .post()
                .uri("v1/otp/generate")
                .contentType(MediaType.APPLICATION_JSON)
                .bodyValue(generateRequestOtp())
                .retrieve()
                .bodyToMono(OtpResponse.class)
                .map(this::mapeoOtp)
                .block();

        String xml;
        try {
            criteria = setearOtp(criteria,otp);
            xml = xmlMapper.writeValueAsString(criteria);
        } catch (JsonProcessingException ex){
            log.error("ocurrio un error en OTPServiceAdapter : " + ex);
            return new ResponseOTP();
        }
        log.info(xml);
        return xmlWebClient
                .post()
                .uri("/XTMail")
                .contentType(MediaType.TEXT_XML)
                .bodyValue(xml)
                .retrieve()
                .bodyToMono(String.class)
                .map(this::mapeoResponse)
                .block();

    }

    private ResponseOTP mapeoResponse(String valueXml){
        log.info(valueXml);
        try {
            return xmlMapper.readValue(valueXml, ResponseOTP.class);
        } catch (JsonProcessingException e) {
            log.error("ocurrio un error al trasnformar el response : "+ e);
            return new ResponseOTP();
        }
    }

    private RequestOTP setearOtp(RequestOTP criteria, String otp){
        String cadenaOtp = criteria.getRecipient().getPersonalization().get(0).getVALUE();
        cadenaOtp = cadenaOtp + otp;

        criteria.getRecipient().getPersonalization().get(0).setVALUE(cadenaOtp);

        return criteria;
    }

    private OtpRequest generateRequestOtp(){
        return new OtpRequest(UUID.randomUUID().toString());
    }

    private String mapeoOtp(OtpResponse otpResponse) {
        return otpResponse.getData().getOtpCode();
    }

    private static WrapperClassifyEmail generateResponsseValidCode(StatusInfo.StatusCodeEnum status){
        var response = new WrapperClassifyEmail();
        response.setEmail(new Email(null,null,null,null,null,null,null,null,null,new StatusInfo(status)));
        return response;
    }

}

import com.santander.cpe.nxhbsc.cpenxhbscbeemailboxes.infrastructure.constants.Constant;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Service;
import org.springframework.web.reactive.function.BodyInserters;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.core.scheduler.Schedulers;

import java.time.Instant;
import java.time.LocalDateTime;
import java.time.ZoneId;

import static com.santander.cpe.nxhbsc.cpenxhbscbeemailboxes.infrastructure.constants.Constant.MINUTES;


/**
 * The class representative TOken service
 *
 * @author Create by team Proyecto [NUBE] Pe
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class TokenServiceJson {

    private final WebClient otpTokenWebClient;
    private String cachedToken;
    private LocalDateTime tokenExpiry;


  /**
   * Obteing token
   *
   * @return A {@link String} containing token.
   */
  public String getToken(){
    if(cachedToken == null || LocalDateTime.now().isBefore(tokenExpiry)) {
        refreshToken();
    }
    return cachedToken;
  }

  private synchronized void refreshToken(){
    if(cachedToken!= null && LocalDateTime.now().isBefore(tokenExpiry)){
        return;
    }

      OTPTokenResponse response = otpTokenWebClient.post()
            .uri("oauth/token")
            .headers(httpHeaders -> httpHeaders.setBasicAuth(Constant.CLIENT_VALUE, Constant.CLIENT_SECRET))
            .contentType(MediaType.APPLICATION_FORM_URLENCODED)
            .body(BodyInserters
                    .fromFormData(Constant.GRANT_TYPE,Constant.GRANT_TYPE_OTP_VALUE)
                    .with(Constant.SCOPE,Constant.SCOPE_VALUE))
            .retrieve()
            .bodyToMono(OTPTokenResponse.class)
            .subscribeOn(Schedulers.boundedElastic())
            .block();
    if(response == null || response.getData() == null ){
        throw new RuntimeException("Failed to get token: ");
    }
    cachedToken = response.getData().getAccessToken();
    tokenExpiry = Instant.ofEpochMilli(response.getData().getExpiresIn())
            .atZone(ZoneId.systemDefault())
            .toLocalDateTime()
            .minusSeconds(MINUTES);
  }


}

                                    .addHandlerLast(
                                            new WriteTimeoutHandler(
                                                    TIMEOUT, TimeUnit.MILLISECONDS)));

            .clientConnector(new ReactorClientHttpConnector(httpClient))
            .baseUrl("https://otpsantander.celmediamobile.pe/api")
            .filter(addTokenFilter(tokenServiceJson))
            .defaultHeader(HttpHeaders.ACCEPT, MediaType.APPLICATION_XML_VALUE)
            .build();
