package com.santander.scp.pltint.scppltintbuspartiesperv1.infrastructure.adapters.output.dynamodb;
 
import com.santander.framework.springboot.core.exceptions.NotFoundException;

import com.santander.scp.pltint.scppltintbuspartiesperv1.application.ports.output.UpdatePartyDynamoDBOutputPort;

import com.santander.scp.pltint.scppltintbuspartiesperv1.infrastructure.adapters.input.rest.data.WrapperPostPartyRequest;

import com.santander.scp.pltint.scppltintbuspartiesperv1.infrastructure.adapters.output.dynamodb.mapper.UpdatePartyMapper;

import com.santander.scp.pltint.scppltintbuspartiesperv1.infrastructure.adapters.output.dynamodb.models.PartyDetailEntity;

import java.time.Instant;

import org.springframework.beans.factory.annotation.Value;

import org.springframework.stereotype.Component;

import software.amazon.awssdk.enhanced.dynamodb.DynamoDbEnhancedClient;

import software.amazon.awssdk.enhanced.dynamodb.DynamoDbTable;

import software.amazon.awssdk.enhanced.dynamodb.Key;

import software.amazon.awssdk.enhanced.dynamodb.TableSchema;

import software.amazon.awssdk.enhanced.dynamodb.model.PutItemEnhancedRequest;
 
/**

* Adapter implementation for updating party details in DynamoDB. This class interacts with DynamoDB

* to perform update operations on party details.

*/

@Component

public class UpdatePartyDynamoDBAdapter implements UpdatePartyDynamoDBOutputPort {
 
  private final UpdatePartyMapper updatePartyMapper;

  private final DynamoDbEnhancedClient dynamoDbEnhancedClient;
 
  /**

   * Constructor for `UpdatePartyDynamoDBAdapter`.

   *

   * @param updatePartyMapper      The mapper used to convert and merge party details.

   * @param dynamoDbEnhancedClient The DynamoDB enhanced client for interacting with the database.

   */

  public UpdatePartyDynamoDBAdapter(

      UpdatePartyMapper updatePartyMapper,

      DynamoDbEnhancedClient dynamoDbEnhancedClient

  ) {

    this.dynamoDbEnhancedClient = dynamoDbEnhancedClient;

    this.updatePartyMapper = updatePartyMapper;

  }
 
  @Value("${apparsenal.backing-services.aws.dynamodb.tables.party-detail.name}")

  private String tableName;
 
  /**

   * Updates the details of a party in the DynamoDB table.

   *

   * @param user                    The user performing the update operation.

   * @param partyId                 The unique identifier of the party to be updated.

   * @param wrapperPostPartyRequest The request object containing the updated party details.

   * @throws NotFoundException if the party with the given ID is not found in the database.

   */

  @Override

  public void updateParty(String user, String partyId,

      WrapperPostPartyRequest wrapperPostPartyRequest) {

    if (wrapperPostPartyRequest == null) {

      return;

    }

    // Retrieve the DynamoDB table for party details

    DynamoDbTable<PartyDetailEntity> partyTable = dynamoDbEnhancedClient

        .table(tableName, TableSchema.fromBean(PartyDetailEntity.class));
 
    // Fetch the existing party details using the party ID

    PartyDetailEntity entity = partyTable.getItem(

        Key.builder().partitionValue(partyId).build());
 
    // Throw an exception if the party is not found

    if (entity == null) {

      throw new NotFoundException("parties.not_found");

    }
 
    // Map the request object to a new entity and merge it with the existing entity

    PartyDetailEntity newEntity = updatePartyMapper.toPartyDetailEntity(wrapperPostPartyRequest);

    updatePartyMapper.mergePartyDetailEntity(entity, newEntity);
 
    // Update the metadata fields

    entity.setUpdatedAt(Instant.now());

    entity.setUpdatedUser(user);
 
    // Save the updated entity back to the DynamoDB table

    partyTable.putItem(

        PutItemEnhancedRequest.builder(PartyDetailEntity.class)

            .item(entity)

            .build());

  }

}
 
package com.santander.scp.pltint.scppltintbuspartiesperv1.infrastructure.adapters.output.dynamodb.config;
 
import org.springframework.beans.factory.annotation.Qualifier;

import org.springframework.beans.factory.annotation.Value;

import org.springframework.context.annotation.Bean;

import org.springframework.context.annotation.Configuration;

import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;

import software.amazon.awssdk.enhanced.dynamodb.DynamoDbEnhancedClient;

import software.amazon.awssdk.regions.Region;

import software.amazon.awssdk.services.dynamodb.DynamoDbClient;
 
/**

* Configuration class for setting up DynamoDB clients.

* <p>

* This class provides beans for the DynamoDbClient and DynamoDbEnhancedClient, using AWS region and

* credentials from environment variables. It is used to enable integration with AWS DynamoDB in the

* application context.

*/

@Configuration

public class DynamoDBConfig {
 
  @Value("${apparsenal.backing-services.aws.region}")

  private String awsRegion;
 
  /**

   * Creates and configures a DynamoDbClient bean.

   *

   * @return a configured DynamoDbClient instance

   */

  @Bean(name = "dynamoDbClient")

  public DynamoDbClient dynamoDbClient() {

    return DynamoDbClient.builder()

        .credentialsProvider(DefaultCredentialsProvider.create())

        .region(Region.of(awsRegion))

        .build();

  }
 
  /**

   * Creates a DynamoDbEnhancedClient bean using the provided DynamoDbClient.

   *

   * @param dynamoDbClient the DynamoDbClient to use

   * @return a configured DynamoDbEnhancedClient instance

   */

  @Bean

  public DynamoDbEnhancedClient dynamoDbEnhancedClient(

      @Qualifier("dynamoDbClient") DynamoDbClient dynamoDbClient) {

    return DynamoDbEnhancedClient.builder()

        .dynamoDbClient(dynamoDbClient)

        .build();

  }

}
 
package com.santander.scp.pltint.scppltintbuspartiesperv1.infrastructure.adapters.output.dynamodb.config;
 
import org.jetbrains.annotations.Nullable;

import software.amazon.awssdk.enhanced.dynamodb.*;

import software.amazon.awssdk.enhanced.dynamodb.model.PutItemEnhancedRequest;

import software.amazon.awssdk.enhanced.dynamodb.Expression;

import software.amazon.awssdk.enhanced.dynamodb.model.ScanEnhancedRequest;

import software.amazon.awssdk.services.dynamodb.model.AttributeValue;
 
import java.util.*;

import java.util.stream.Collectors;
 
public abstract class BaseDynamoRepository<T> {
 
    protected final DynamoDbEnhancedClient enhancedClient;

    protected final DynamoDbTable<T> table;

    private final String partitionKeyAttr;

    @Nullable

    private final String sortKeyAttr;

    private final Class<T> itemClass;
 
    protected BaseDynamoRepository(DynamoDbEnhancedClient enhancedClient,

                                   String tableName,

                                   TableSchema<T> schema,

                                   Class<T> itemClass,

                                   String partitionKeyAttr,

                                   @Nullable String sortKeyAttr) {

        this.enhancedClient = enhancedClient;

        this.table = enhancedClient.table(tableName, schema);

        this.itemClass = itemClass;

        this.partitionKeyAttr = partitionKeyAttr;

        this.sortKeyAttr = sortKeyAttr;

    }
 
    public void putIfAbsent(T item) {

        String expr = "attribute_not_exists(" + partitionKeyAttr + ")";

        if (sortKeyAttr != null && !sortKeyAttr.isBlank()) {

            expr += " AND attribute_not_exists(" + sortKeyAttr + ")";

        }

        Expression condition = Expression.builder()

                .expression(expr)

                .build();

        PutItemEnhancedRequest<T> request = PutItemEnhancedRequest.builder(itemClass)

                .item(item)

                .conditionExpression(condition)

                .build();

        table.putItem(request);

    }
 
    public Optional<T> getByKey(String partitionKeyValue, @Nullable String sortKeyValue) {

        Key key = (sortKeyAttr == null || sortKeyValue == null || sortKeyValue.isBlank())

                ? Key.builder().partitionValue(partitionKeyValue).build()

                : Key.builder().partitionValue(partitionKeyValue).sortValue(sortKeyValue).build();
 
        T item = table.getItem(r -> r.key(key));

        return Optional.ofNullable(item);

    }
 
 
    public List<T> findAll() {

        List<T> results = new ArrayList<>();

        table.scan(ScanEnhancedRequest.builder().build())

                .items()

                .forEach(results::add);

        return results;

    }
 
}
