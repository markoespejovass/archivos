    @Test
    void uploadDocument_ok() {

        WrapperPostDocumentsRequest request = buildRequest();

        FileEntity entity = new FileEntity();
        entity.setDocumentNumber("123");

        WrapperPostDocumentsResponse response =
                new WrapperPostDocumentsResponse();

        when(mapper.toEntity(any(), anyString(), anyString()))
                .thenReturn(entity);

        when(repository.putIfAbsent(any(FileEntity.class)))
                .thenReturn(entity);

        when(mapper.toDTOUpload(entity))
                .thenReturn(response);

        WrapperPostDocumentsResponse result =
                adapter.uploadDocument(request);

        assertNotNull(result);

        verify(s3Client).putObject(
                any(PutObjectRequest.class),
                any(RequestBody.class));

        verify(repository).putIfAbsent(any(FileEntity.class));
        verify(mapper).toDTOUpload(entity);
    }

    @Test
    void downloadDocument_ok() throws Exception {

        WrapperGetDocumentsRequest request =
                new WrapperGetDocumentsRequest();

        WrapperMyDocumentResponse document =
                new WrapperMyDocumentResponse();

        document.setDocumentId("123");

        request.setDocument(document);

        FileEntity entity = new FileEntity();
        entity.setUrl("folder/file.pdf");

        WrapperGetDocumentsResponse response =
                new WrapperGetDocumentsResponse();

        PresignedGetObjectRequest presigned =
                mock(PresignedGetObjectRequest.class);

        when(repository.getByKey("123", null))
                .thenReturn(Optional.of(entity));

        when(presigned.url())
                .thenReturn(new URL("https://test.com/file.pdf"));

        when(s3Presigner.presignGetObject(any()))
                .thenReturn(presigned);

        when(mapper.donwloadMapper(anyString()))
                .thenReturn(response);

        WrapperGetDocumentsResponse result =
                adapter.downloadDocument(request);

        assertNotNull(result);

        verify(mapper).donwloadMapper(anyString());
    }

    // =========================
    // builders
    // =========================

    private WrapperPostDocumentsRequest buildRequest() {

        String base64 = Base64.getEncoder()
                .encodeToString("contenido".getBytes());

        WrapperMySearchDocumentResponseOwnersInner owner =
                new WrapperMySearchDocumentResponseOwnersInner();

        owner.setOwnerId("12345678");

        WrapperMyDocumentRequest document =
                new WrapperMyDocumentRequest();

        document.setFolderReference(base64);
        document.setAclReference("ACL");
        document.setMimeType("application/pdf");
        document.setName("file.pdf");
        document.setOwners(List.of(owner));

        WrapperCallerInformation caller =
                new WrapperCallerInformation();

        caller.setAppId("APP");

        WrapperPostDocumentsRequest request =
                new WrapperPostDocumentsRequest();

        request.setDocument(document);
        request.setCallerInformation(caller);

        return request;
    }

    private WrapperPostSearchDocumentsRequest buildSearchRequest() {

        WrapperMySearchFilterRequestSearchPropertiesInner property =
                new WrapperMySearchFilterRequestSearchPropertiesInner();

        property.setKeyId("DOCUMENT_NUMBER");
        property.setKeyValue("123");
        property.setKeyOperator(
                WrapperMySearchFilterRequestSearchPropertiesInner
                        .KeyOperatorEnum.EQ);

        WrapperMySearchFilterRequest filter =
                new WrapperMySearchFilterRequest();

        filter.setDiscriminator("AND");
        filter.setSearchProperties(List.of(property));

        WrapperPostSearchDocumentsRequest request =
                new WrapperPostSearchDocumentsRequest();

        request.setSearchParameters(filter);

        return request;
    }
