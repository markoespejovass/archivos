package com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.output.jpa;


import com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.adapters.input.rest.data.WrapperMySearchFilterRequestSearchPropertiesInner;
import jakarta.persistence.criteria.Predicate;
import org.springframework.data.jpa.domain.Specification;

import java.util.ArrayList;
import java.util.List;

import static com.santander.cpe.nxhbsc.cpenxhbscbedocmanagement.infrastructure.constants.Constant.*;

public class FileSpecificationBuilder {

    public static Specification<FileEntity> build(
            List<WrapperMySearchFilterRequestSearchPropertiesInner> filters,
            String discriminator
    ) {

        return (root, query, cb) -> {

            List<Predicate> predicates = new ArrayList<>();

            for (WrapperMySearchFilterRequestSearchPropertiesInner filter : filters) {

                String field = mapField(filter.getKeyId());
                String value = filter.getKeyValue();

                switch (filter.getKeyOperator().getValue()) {
                    case CONTAINS:
                        predicates.add(
                                cb.like(
                                        cb.lower(root.get(field)),
                                        "%" + value.toLowerCase() + "%"
                                )
                        );
                        break;

                    case EQUAL:
                        predicates.add(
                                cb.equal(root.get(field), castValue(field, value))
                        );
                        break;
                }
            }

            // AND por defecto
            if ("OR".equalsIgnoreCase(discriminator)) {
                return cb.or(predicates.toArray(new Predicate[0]));
            }

            return cb.and(predicates.toArray(new Predicate[0]));
        };

    }


    private static String mapField(String keyId) {
        return switch (keyId) {
            case NAME -> "name";
            case MIME_TYPE -> "mimeType";
            case STATUS -> "status";
            case VERSION -> "version";
            case ID -> "id";
            default -> throw new IllegalArgumentException("Invalid field");
        };
    }


    private static Object castValue(String field, String value) {
        return switch (field) {
            case "version" -> Integer.valueOf(value);
            case "id" -> Long.valueOf(value);
            default -> value;
        };
    }
}
