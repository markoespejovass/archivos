package com.example.demo.infrastructure.adapter.out.external.client;

import com.example.demo.domain.model.otp.sendMailing.RequestOTP;
import com.example.demo.domain.model.otp.sendMailing.ResponseOTP;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.dataformat.xml.XmlMapper;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;

import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;

import org.springframework.http.MediaType;
import org.springframework.web.reactive.function.client.WebClient;

import reactor.core.publisher.Mono;

import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
class OTPServiceAdapterTest {

    @Mock
    private WebClient webClient;

    @Mock
    private XmlMapper xmlMapper;

    @Mock
    private WebClient.RequestBodyUriSpec requestBodyUriSpec;

    @Mock
    private WebClient.RequestBodySpec requestBodySpec;

    @Mock
    private WebClient.RequestHeadersSpec<?> requestHeadersSpec;

    @Mock
    private WebClient.ResponseSpec responseSpec;

    @InjectMocks
    private OTPServiceAdapter adapter;

    // =========================
    // ✅ SUCCESS CASE
    // =========================
    @Test
    void shouldSendMailingSuccessfully() throws Exception {

        RequestOTP request = new RequestOTP();
        String xmlRequest = "<xml>request</xml>";
        String xmlResponse = "<xml>response</xml>";

        ResponseOTP expectedResponse = new ResponseOTP();

        // Mock XML serialization
        when(xmlMapper.writeValueAsString(request)).thenReturn(xmlRequest);

        // Mock WebClient chain
        when(webClient.post()).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.uri("/XTMail")).thenReturn(requestBodySpec);
        when(requestBodySpec.contentType(MediaType.TEXT_XML)).thenReturn(requestBodySpec);
        when(requestBodySpec.bodyValue(xmlRequest)).thenReturn(requestHeadersSpec);
        when(requestHeadersSpec.retrieve()).thenReturn(responseSpec);
        when(responseSpec.bodyToMono(String.class)).thenReturn(Mono.just(xmlResponse));

        // Mock XML deserialization
        when(xmlMapper.readValue(xmlResponse, ResponseOTP.class))
                .thenReturn(expectedResponse);

        // Act
        ResponseOTP result = adapter.sendMailing(request);

        // Assert
        assertNotNull(result);
        assertEquals(expectedResponse, result);

        verify(xmlMapper).writeValueAsString(request);
        verify(xmlMapper).readValue(xmlResponse, ResponseOTP.class);

        verify(webClient).post();
        verify(requestBodyUriSpec).uri("/XTMail");
        verify(requestBodySpec).contentType(MediaType.TEXT_XML);
        verify(requestBodySpec).bodyValue(xmlRequest);
        verify(requestHeadersSpec).retrieve();
        verify(responseSpec).bodyToMono(String.class);
    }

    // =========================
    // ❌ ERROR PARSING XML
    // =========================
    @Test
    void shouldReturnEmptyResponseWhenXmlParsingFails() throws Exception {

        RequestOTP request = new RequestOTP();
        String xmlRequest = "<xml>request</xml>";
        String invalidXml = "INVALID_XML";

        when(xmlMapper.writeValueAsString(request)).thenReturn(xmlRequest);

        when(webClient.post()).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.uri(anyString())).thenReturn(requestBodySpec);
        when(requestBodySpec.contentType(any())).thenReturn(requestBodySpec);
        when(requestBodySpec.bodyValue(any())).thenReturn(requestHeadersSpec);
        when(requestHeadersSpec.retrieve()).thenReturn(responseSpec);
        when(responseSpec.bodyToMono(String.class)).thenReturn(Mono.just(invalidXml));

        when(xmlMapper.readValue(invalidXml, ResponseOTP.class))
                .thenThrow(new JsonProcessingException("error") {});

        // Act
        ResponseOTP result = adapter.sendMailing(request);

        // Assert
        assertNotNull(result); // retorna objeto vacío según tu lógica
    }

    // =========================
    // ❌ ERROR EN WEBCLIENT
    // =========================
    @Test
    void shouldThrowExceptionWhenWebClientFails() throws Exception {

        RequestOTP request = new RequestOTP();

        when(xmlMapper.writeValueAsString(request)).thenReturn("<xml/>");

        when(webClient.post()).thenReturn(requestBodyUriSpec);
        when(requestBodyUriSpec.uri(anyString())).thenReturn(requestBodySpec);
        when(requestBodySpec.contentType(any())).thenReturn(requestBodySpec);
        when(requestBodySpec.bodyValue(any())).thenReturn(requestHeadersSpec);
        when(requestHeadersSpec.retrieve()).thenReturn(responseSpec);

        when(responseSpec.bodyToMono(String.class))
                .thenReturn(Mono.error(new RuntimeException("API error")));

        assertThrows(RuntimeException.class,
                () -> adapter.sendMailing(request));
    }
}
