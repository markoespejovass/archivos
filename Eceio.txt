import org.springframework.data.jpa.domain.Specification;

public class FileSpecificationBuilder {

    public static Specification<FileEntity> build(
            List<SearchProperty> filters,
            String discriminator
    ) {

        return (root, query, cb) -> {

            List<Predicate> predicates = new ArrayList<>();

            for (SearchProperty filter : filters) {

                String field = mapField(filter.getKeyId());
                String value = filter.getKeyValue();

                switch (filter.getKeyOperator()) {

                    case CONTAINS:
                        predicates.add(
                            cb.like(
                                cb.lower(root.get(field)),
                                "%" + value.toLowerCase() + "%"
                            )
                        );
                        break;

                    case EQUAL:
                        predicates.add(
                            cb.equal(root.get(field), castValue(field, value))
                        );
                        break;
                }
            }

            // AND por defecto
            if ("OR".equalsIgnoreCase(discriminator)) {
                return cb.or(predicates.toArray(new Predicate[0]));
            }

            return cb.and(predicates.toArray(new Predicate[0]));
        };
    }
