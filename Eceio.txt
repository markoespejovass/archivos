import org.junit.jupiter.api.Test;
import org.springframework.http.HttpHeaders;
import org.springframework.web.reactive.function.client.*;
import reactor.core.publisher.Mono;
import reactor.test.StepVerifier;

import java.lang.reflect.Method;
import java.net.URI;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

class XmlWebClientConfigFilterTest {

    @Test
    void shouldAddAuthorizationHeader() throws Exception {

        // Mock TokenService
        TokenService tokenService = mock(TokenService.class);
        when(tokenService.getToken()).thenReturn("my-token");

        // Obtener método privado con reflection
        Method method = XmlWebClientConfig.class
                .getDeclaredMethod("addTokenFilter", TokenService.class);

        method.setAccessible(true);

        ExchangeFilterFunction filter =
                (ExchangeFilterFunction) method.invoke(null, tokenService);

        // Request base
        ClientRequest request = ClientRequest
                .create(org.springframework.http.HttpMethod.GET, URI.create("http://test"))
                .build();

        // Aplicar filtro
        Mono<ClientRequest> result =
                filter.filter(request, exchange -> Mono.empty());

        StepVerifier.create(result)
                .assertNext(req -> {
                    String authHeader = req.headers().getFirst(HttpHeaders.AUTHORIZATION);
                    assertEquals("Bearer my-token", authHeader);
                })
                .verifyComplete();
    }
}
