 *
 * @author Create by team Proyecto [NUBE] Pe

@Component
@RequiredArgsConstructor
public class KnowYourCustomerAdapter implements KnowYourCustomerRepositoryPort {

  private final KnokYourCustomerRepository repository;
  private final WeightRepository weightRepository;
  private final KYCEntityMapper mapper;
  private final WeightMapper mapperWeight;
  private final ValueRepository valueRepository;
  private final InputDefinitionRepository inputDefinitionRepository;


  @Value("${apparsenal.backing-services.aws.dynamodb.tables.party-detail.name}")
  private String tableName;

  @Override
  public WrapperPostKycResolutionResponse createProcesstKyc(
      WrapperPostKycResolutionRequest criteria) {

    Kyc kyc = repository.putIfAbsent(
            mapper.WrapperPostKycResolutionRequestToKycEntity(criteria));
    if (Strings.isBlank(kyc.getId()) ){
      return new WrapperPostKycResolutionResponse();
    }
    String id = kyc.getDocumentNumber();
    return mapper.toDto(id);

  }

  @Override
  public Kyc getKyc(String documento){
    // Retrieve the DynamoDB table for party details
    return repository.getBySortKey(documento+"01").get();
  }

  @Override
  public List<ValueMapping> getValues() {
    return valueRepository.findAll();
  }

  @Override
  public List<InputDefinition> getAllInputs() {
    return inputDefinitionRepository.findAll();
  }

  @Override
  public List<WeightRule> getRulesByFactor(String factor) {
    return mapperWeight.toDTOList(weightRepository.findBySortKey(factor));
  }
}

@ExtendWith(MockitoExtension.class)
class KnowYourCustomerAdapterTest {

    @Mock
    private WeightRepository weightRepository;
    @Mock
    private KYCEntityMapper mapper;
    @Mock
    private WeightMapper mapperWeight;
    @Mock
    private ValueRepository valueRepository;
    @Mock
    private InputDefinitionRepository inputDefinitionRepository;
    @Mock
    private DynamoDbEnhancedClient dynamoDbEnhancedClient;

    @Mock
    private DynamoDbTable<Kyc> dynamoDbTable;

    @InjectMocks
    private KnowYourCustomerAdapter adapter;

    @BeforeEach
    void setup() {
        // inyectar nombre de tabla manualmente
        ReflectionTestUtils.setField(adapter, "tableName", "test-table");
    }

    // DYnamo test

    private WrapperPostKycResolutionRequest buildRequest() {
        WrapperPostKycResolutionRequest request = mock(WrapperPostKycResolutionRequest.class);
        var party = mock(Party.class);

        when(request.getParty()).thenReturn(party);
        when(party.getPartyId()).thenReturn("123");

        return request;
    }

    @Test
    void shouldCreateKycSuccessfully() {
        var request = buildRequest();
        Kyc newEntity = new Kyc();

        when(dynamoDbEnhancedClient.table(anyString(), any(TableSchema.class)))
                .thenReturn(dynamoDbTable);

        when(mapper.WrapperPostKycResolutionRequestToKycEntity(request))
                .thenReturn(newEntity);

        when(mapper.toDto(anyString()))
                .thenReturn(new WrapperPostKycResolutionResponse());

        var response = adapter.createProcesstKyc(request);

        assertNotNull(response);
        verify(dynamoDbTable).putItem(any(PutItemEnhancedRequest.class));
    }


    @Test
    void shouldGetKyc() {
        when(dynamoDbEnhancedClient.table(anyString(), any(TableSchema.class)))
                .thenReturn(dynamoDbTable);

        Kyc kyc = new Kyc();

        when(dynamoDbTable.getItem(any(Key.class)))
                .thenReturn(kyc);

        var result = adapter.getKyc("123");

        assertEquals(kyc, result);
    }


    //

    @Test
    void shouldGetValues() {
        when(valueRepository.findAll()).thenReturn(List.of(new ValueMapping()));

        assertFalse(adapter.getValues().isEmpty());
    }

    @Test
    void shouldGetInputs() {
        when(inputDefinitionRepository.findAll()).thenReturn(List.of(new InputDefinition()));

        assertFalse(adapter.getAllInputs().isEmpty());
    }

    @Test
    void shouldGetRulesByFactor() {
        when(weightRepository.findBySortKey("F1")).thenReturn(List.of());
        when(mapperWeight.toDTOList(any())).thenReturn(List.of(new WeightRule()));

        assertFalse(adapter.getRulesByFactor("F1").isEmpty());
    }
}
