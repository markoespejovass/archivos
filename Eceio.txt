private static String mapField(KeyId keyId) {
    switch (keyId) {
        case NAME: return "name";
        case MIME_TYPE: return "mimeType";
        case STATUS: return "status";
        case VERSION: return "version";
        case ID: return "id";
        default: throw new IllegalArgumentException("Invalid field");
    }
}



private static Object castValue(String field, String value) {

    switch (field) {
        case "version":
            return Integer.parseInt(value);

        case "id":
            return Long.parseLong(value);

        default:
            return value;
    }
}
