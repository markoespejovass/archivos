when(responseSpec.onStatus(any(), any())).thenAnswer(invocation -> {

    var function = invocation.getArgument(1);

    var clientResponse = mock(ClientResponse.class);

    when(clientResponse.bodyToMono(String.class))
            .thenReturn(Mono.just("error-body"));

    Mono<?> result = (Mono<?>) function.apply(clientResponse);

    StepVerifier.create(result).verifyError();

    return responseSpec;
});
