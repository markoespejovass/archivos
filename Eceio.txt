@Slf4j
@Service
public class TokenService {

  private final WebClient webClient;

  public TokenService( WebClient.Builder webClient) {
    this.webClient = webClient.baseUrl(Constant.baseURL).build();
  }

    /**
     * Genera un token
     *
     *
     * @return un token generado
     */
  public Mono<String> getToken() {
      return webClient
              .post()
              .uri(Constant.URI_API_TOKEN)
              .contentType(MediaType.APPLICATION_FORM_URLENCODED)
              .body(BodyInserters
                      .fromFormData(Constant.grantType,Constant.grantTypeValue)
                      .with(Constant.clientId,Constant.clientIdValue)
                      .with(Constant.clientSecret,Constant.clientSecretValue)
                      .with(Constant.refreshToken,Constant.refreshTokenValue))
              .retrieve()
              .onStatus(
                      httpStatus -> httpStatus.is4xxClientError() || httpStatus.is5xxServerError(),
                      clientResponse ->
                              clientResponse
                                      .bodyToMono(String.class)
                                      .flatMap(
                                              errorBody -> {
                                                  log.error("Error getting token: {}", errorBody);
                                                  return Mono.error(
                                                          new RuntimeException("Failed to get token: " + errorBody));
                                              }))
              .bodyToMono(TokenResponse.class)
              .map(TokenResponse::getAccessToken)
              .doOnSuccess(token -> log.info("Token obtained successfully"))
              .doOnError(error -> log.error("Error obtaining token: ", error));

  }

}



package com.santander.cpe.nxhbsc.cpenxhbscbeemailboxes.infrastructure.config;

import com.santander.cpe.nxhbsc.cpenxhbscbeemailboxes.infrastructure.adapters.output.TokenService;
import io.netty.handler.ssl.SslContextBuilder;
import io.netty.handler.ssl.util.InsecureTrustManagerFactory;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpHeaders;
import org.springframework.http.MediaType;
import org.springframework.http.client.reactive.ReactorClientHttpConnector;
import org.springframework.http.codec.xml.Jaxb2XmlDecoder;
import org.springframework.http.codec.xml.Jaxb2XmlEncoder;
import org.springframework.web.reactive.function.client.ClientRequest;
import org.springframework.web.reactive.function.client.ExchangeFilterFunction;
import org.springframework.web.reactive.function.client.ExchangeStrategies;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.netty.http.client.HttpClient;

import javax.net.ssl.SSLException;

/**
 * The class representative Xml config
 *
 * @author Create by team Proyecto [NUBE] Pe
 */
@Configuration public class XmlWebClientConfig {
@Bean
public WebClient xmlWebClient(TokenService tokenService) {

    ExchangeStrategies strategies = ExchangeStrategies.builder()
            .codecs(configurer -> {
                configurer.defaultCodecs()
                        .jaxb2Encoder(new Jaxb2XmlEncoder());
                configurer.defaultCodecs()
                        .jaxb2Decoder(new Jaxb2XmlDecoder());
            })
            .build();
    HttpClient httpClient = HttpClient.create()
            .secure(t -> {
                try {
                    t.sslContext(SslContextBuilder.forClient().trustManager(
                            InsecureTrustManagerFactory.INSTANCE
                    ).build());
                } catch (SSLException e) {
                    throw new RuntimeException(e);
                }
            });

    return WebClient.builder()
            .clientConnector(new ReactorClientHttpConnector(httpClient))
            .baseUrl("http://transact-campaign-us-2.goacoustic.com")
            .exchangeStrategies(strategies)
            .filter(addTokenFilter(tokenService))
            .defaultHeader(HttpHeaders.ACCEPT, MediaType.APPLICATION_XML_VALUE)
            .build();
    }

    private ExchangeFilterFunction addTokenFilter(TokenService tokenService) {
        return (request, next) ->
                tokenService
                        .getToken()
                        .flatMap(
                                token -> {
                                    ClientRequest authorizedRequest =
                                            ClientRequest.from(request)
                                                    .header(HttpHeaders.AUTHORIZATION, "Bearer " + token)
                                                    .build();
                                    return next.exchange(authorizedRequest);
                                });
    }
}


package com.santander.cpe.nxhbsc.cpenxhbscbeemailboxes.infrastructure.util;

import java.util.concurrent.ThreadLocalRandom;

import static com.santander.cpe.nxhbsc.cpenxhbscbeemailboxes.infrastructure.constants.Constant.temp_2;

/**
 * The class representative Utils otp
 *
 * @author Create by team Proyecto [NUBE] Pe
 */
public class UtilsOtp {
    private UtilsOtp() {
        // Evita instanciación
    }

    /**
     * Genera un número par aleatorio de 8 dígitos válido (empieza de 1-9)
     *
     * @return el valor del numero generado
     */
    public static String generarNumeroPar8Digitos() {
        var numero = ThreadLocalRandom.current().nextInt(10_000_000, 100_000_000);

        // asegurar que sea par
        if (numero % 2 != 0) {
            numero++;
        }

        // en el caso extremadamente raro de pasar a 100000000
        if (numero > 99_999_999) {
            numero -= 2;
        }

        return String.valueOf(numero);
    }

    /**
     * Valida si un número es par y tiene 8 dígitos válidos (1-9 al inicio)
     *
     * @return validar el numero generado
     */
    public static boolean esNumeroPar8DigitosValido(String numero) {

        if (numero == null) {
            return false;
        }

        // Debe tener exactamente 8 dígitos y empezar 1-9
        if (!numero.matches("^[1-9]\\d{7}$")) {
            return false;
        }

        var valor = Integer.parseInt(numero);

        return valor % temp_2 == 0;
    }
}
