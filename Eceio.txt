package com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.output.jpa;

import com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.application.ports.output.ChannelServiceRepositoryPort;
import com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.input.rest.data.WrapperCustomerPositionResponse;
import com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.output.jpa.mapper.ProductJPAMapper;
import org.springframework.stereotype.Component;

import com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.output.jpa.repository.ExampleUserRepository;

/**
 * User management output adapter that saves info using JPA repository
 * 
 * @author Created by Team DCOE Mx
 */
@Component
public class UserManagementJPAAdapter implements ChannelServiceRepositoryPort {

    private final ExampleUserRepository userDataRepository;

    public UserManagementJPAAdapter(ExampleUserRepository userDataRepository) {
        this.userDataRepository = userDataRepository;
    }



    @Override
    public WrapperCustomerPositionResponse getListServices(String customerId) {
        return ProductJPAMapper.toDTO(userDataRepository.findByNroDocumento(convertToDni(customerId)));
    }

    private String convertToDni(String customerId){
        return customerId.substring(0,customerId.length()-2);
    }

}



package com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.output.jpa.mapper;

import com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.input.rest.data.*;
import com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.output.jpa.data.Product;

import java.util.List;
import java.util.stream.Collectors;


public class ProductJPAMapper {



    public static WrapperCustomerPositionResponse toDTO(List<Product> entities){
        WrapperCustomerPositionResponse response = new WrapperCustomerPositionResponse();
        WrapperCardsInformation creditsCards = new WrapperCardsInformation();

        List<WrapperCardWithLinks> cards = entities.stream().map(enty ->{
            WrapperCardWithLinks wrapperCard = new WrapperCardWithLinks();
            Card card = new Card();
            // Datos de campaña
            card.setCardId(enty.getCanal());
            WrapperDebitCardCardIdentification wrapperDebitCardCardIdentification = new WrapperDebitCardCardIdentification();
            wrapperDebitCardCardIdentification.setDisplayNumber(enty.getPeriodo().toString());
            card.setCardIdentification(wrapperDebitCardCardIdentification);
            com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.input.rest.data.Product product = new com.santander.cpe.nxhbsc.cpenxhbscbeproducoffering.infrastructure.adapters.input.rest.data.Product();
            product.setProductCode(enty.getOfertaTasa().toString());
            product.setProductDescription(enty.getPorcDispEfectivo().toString());
            card.setProduct(product);

            // Datos de la tarjeta
            card.setType(enty.getOfertaTipoTc());
            CardCreditLimitLevel cardCreditLimitLevel = new CardCreditLimitLevel();
            cardCreditLimitLevel.setLevelDescription(enty.getOfertaLineaTc().toString());
            card.setCreditLimitLevel(cardCreditLimitLevel);

            //Caracteristicas
            WrapperDebitCardCardholder caracteristicas = new WrapperDebitCardCardholder();
            caracteristicas.setCardholderId(enty.getCaracteristica1());
            caracteristicas.setTypeCode(enty.getCaracteristica2());
            caracteristicas.setTypeDescription(enty.getCaracteristica3());
            //WrapperDebitCardBalancesInner matrices = new WrapperDebitCardBalancesInner();

            WrapperDebitCardStatusInfo segmentos = new WrapperDebitCardStatusInfo();
            segmentos.setStatusCode(enty.getSegmento1());
            segmentos.setStatusDescription(enty.getSegmento2());
            segmentos.setStatusReasonCode(enty.getSegmento3());
            segmentos.setStatusReasonDescription(enty.getSegmento4());

            card.setStatusInfo(segmentos);
            card.setCardholder(caracteristicas);
            wrapperCard.setCard(card);
            return wrapperCard;
        }).collect(Collectors.toList());

        creditsCards.setList(cards);
        response.setCards(creditsCards);
        return response;
    }
}
