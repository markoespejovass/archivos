package com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.config;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.springframework.test.util.ReflectionTestUtils;
import software.amazon.awssdk.services.dynamodb.DynamoDbClient;
import software.amazon.awssdk.enhanced.dynamodb.DynamoDbEnhancedClient;
import software.amazon.awssdk.services.sts.StsClient;

import static org.junit.jupiter.api.Assertions.*;

class DynamoDBConfigTest {

    private DynamoDBConfig dynamoDBConfig;

    @BeforeEach
    void setUp() {
        dynamoDBConfig = new DynamoDBConfig();

        ReflectionTestUtils.setField(
                dynamoDBConfig,
                "awsRegion",
                "us-east-1");

        ReflectionTestUtils.setField(
                dynamoDBConfig,
                "dynamoRoleArn",
                "arn:aws:iam::123456789012:role/test-role");
    }

    @Test
    void shouldCreateStsClient() {
        StsClient stsClient = dynamoDBConfig.stsClient();

        assertNotNull(stsClient);
    }

    @Test
    void shouldCreateDynamoDbClient() {
        StsClient stsClient = dynamoDBConfig.stsClient();

        DynamoDbClient dynamoDbClient =
                dynamoDBConfig.dynamoDbClient(stsClient);

        assertNotNull(dynamoDbClient);
    }

    @Test
    void shouldCreateDynamoDbEnhancedClient() {
        StsClient stsClient = dynamoDBConfig.stsClient();

        DynamoDbClient dynamoDbClient =
                dynamoDBConfig.dynamoDbClient(stsClient);

        DynamoDbEnhancedClient enhancedClient =
                dynamoDBConfig.dynamoDbEnhancedClient(dynamoDbClient);

        assertNotNull(enhancedClient);
    }
}
