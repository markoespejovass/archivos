package com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.mapper;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.json.JsonMapper;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.PoliticalExposure;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.EmploymentInformation;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.Party;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.Person;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionResponse;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.input.rest.data.WrapperPostKycResolutionRequest;
import com.santander.cpe.nxhbsc.cpenxhbscbeknowyocustomer.infrastructure.adapters.output.jpa.data.Kyc;
import org.mapstruct.Mapper;
import org.mapstruct.Mapping;
import org.mapstruct.Named;

/**
 * The class representative Mapper
 *
 * @author Create by team Proyecto [NUBE] Pe
 */
@Mapper(componentModel = "spring")
public interface KYCEntityMapper {

    JsonMapper jsonMapper = new JsonMapper();

    @Mapping(target = "id", ignore = true)
    //factor cliente
    @Mapping(target = "documentNumber", expression = "java(getDocumentNumber(request))")
    @Mapping(target = "typeCustomer", expression = "java(getTC(request))")// tipo de cliente
    @Mapping(target = "typeStructure", expression = "java(getValor(request))")//  tipo de pn
    @Mapping(target = "economicActivityCategory", source = "request.party.person.employmentInformation.economicActivity.categoryDescription")// categoria laboral
    @Mapping(target = "occupationDescription", source = "request.party.person.employmentInformation.occupationDescription")// profesion, valor del sujeto
    @Mapping(target = "valueOfTheSubject", expression = "java(getVS(request))")
    @Mapping(target = "dueDiligenceRegime", expression = "java(getDueDiligenceRegime(request))")// regimen de debida dilgencia
    @Mapping(target = "backgroundOfBlockingList", expression = "java(getBackgroundOfBlockingList(request))")//antecedente de lista bloqueante
    @Mapping(target = "isPep", source = "request.party.person.politicalExposure.isSelfDeclaredPEP")// pep
    @Mapping(target = "birthDate", source = "request.party.person.birthDate")// birthDate
    @Mapping(target = "ownershipStructure", expression = "java(getOwnershipStructure(request))")
    //factor canal
    @Mapping(target = "productDescription", expression = "java(request.getProducts().get(0).getProductDescription())")
    @Mapping(target = "channelDescription", source = "request.preferenceChannel.description")
    @Mapping(target = "transactionality", constant = "Transaccionalidad regular")
    @Mapping(target = "productName", expression = "java(request.getProducts().get(0).getFamily().getProductFamily().getName())")
    //factor zg
    @Mapping(target = "countryOfResidence", source = "request.party.person.countryOfResidence.code")// pais de residensia,
    @Mapping(target = "presenceOfTheGroup", source = "request.party.person.countryOfResidence.code", qualifiedByName = "mapPG")// presencia del grupo
    @Mapping(target = "expCountryRoot", source = "request.party.person.countryOfResidence.code", qualifiedByName = "mapEXP")//exp pais rais
    @Mapping(target = "countryOfBirth", source = "request.party.person.placeOfBirth.country.name")
    @Mapping(target = "stateOfResidence", source = "request.party.person.countryOfResidence.name")// localidad de residencia
    Kyc WrapperPostKycResolutionRequestToKycEntity(WrapperPostKycResolutionRequest request);

    @Mapping(target = "kycResolutionId", source = "id")
    WrapperPostKycResolutionResponse toDto(String id);

    default String getVS(WrapperPostKycResolutionRequest request){
        Party party = request.getParty();
        if(party == null){
            return "5";
        }
        Person person = party.getPerson();
        if (person == null){
            return "5";
        }
        EmploymentInformation employmentInformation = person.getEmploymentInformation();
        if(employmentInformation == null){
            return "5";
        }
        String ocupation = employmentInformation.getOccupationDescription();
        if(ocupation == null){
            return "5";
        }
        if ("5ta Categoría".equals(ocupation)){
            return "1";
        }
        return "5";
    }

    default String getDocumentNumber(WrapperPostKycResolutionRequest request){
        Party party = request.getParty();
        if(party == null){
            return "";
        }
        Person person = party.getPerson();
        if (person == null){
            return "";
        }
        return person.getDocuments().get(0).getDocumentNumber();
    }

    default String getTC(WrapperPostKycResolutionRequest request){
        Party party = request.getParty();
        if(party == null){
            return "";
        }
        Person person = party.getPerson();
        if (person == null){
            return "";
        }
        if(person.getDocuments().get(0).getDocumentNumber().length()<=8){
            return "Persona Natural";
        }
        return "";
    }

    default String getValor(WrapperPostKycResolutionRequest request){
        Party party = request.getParty();
        if(party == null){
            return "";
        }
        Person person = party.getPerson();
        if (person == null){
            return "";
        }
        if(person.getDocuments().get(0).getDocumentNumber().length()<=8){
            return "Física";
        }
        return "";
    }


    default String getDueDiligenceRegime(WrapperPostKycResolutionRequest request){
        Party party = request.getParty();
        if(party == null){
            return "";
        }
        Person person = party.getPerson();
        if (person == null){
            return "";
        }
        PoliticalExposure politicalExposure = person.getPoliticalExposure();
        if(politicalExposure == null){
            return "";
        }
        if("Si".equalsIgnoreCase(politicalExposure.getDeclarationComments()) ||
                politicalExposure.getIsSelfDeclaredPEP()){
            return "Régimen Reforzado";
        }
        return "Régimen General";
    }

    default String getBackgroundOfBlockingList(WrapperPostKycResolutionRequest request){
        Party party = request.getParty();
        if(party == null){
            return "";
        }
        Person person = party.getPerson();
        if (person == null){
            return "";
        }
        PoliticalExposure politicalExposure = person.getPoliticalExposure();
        if(politicalExposure == null){
            return "";
        }
        if("Si".equalsIgnoreCase(politicalExposure.getDeclarationComments())){
            return "SI";
        }
        return "";
    }

    default String getOwnershipStructure(WrapperPostKycResolutionRequest request){
        Party party = request.getParty();
        if(party == null){
            return "";
        }
        Person person = party.getPerson();
        if (person == null){
            return "";
        }
        if(person.getDocuments().get(0).getDocumentNumber().length()<=8){
            return "Simple";
        }
        return "";
    }

    @Named("mapEXP")
    default String getEXP(String countryCode){
        if("PE".equals(countryCode)){
            return "No hay Exposición";
        }
        return "";
    }

    @Named("mapPG")
    default String getPG(String countryCode){
        if("PE".equals(countryCode)){
            return "Nacional";
        }
        return "";
    }

    default String toStringJson(Object o){
        if(o instanceof String){
            return (String)o;
        }
        try {
            return jsonMapper.writeValueAsString(o);
        } catch (JsonProcessingException e) {
            return "Error al parsear objeto";
        }
    }
}
