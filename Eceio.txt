/**
 * @author Create by team Proyecto [NUBE] Pe
 */
@Component
@Slf4j
@RequiredArgsConstructor
public class FacephiAdapter implements FacephiServiceOutputPort {

  private final WebClient webClient;
  private final FacephiMapper mapper;
  private final ObjectMapper objectMapper;
  private final ApiProperties apiProperties;

  private String urlCallback;

  @Override
  public WrapperExecution<UserTemplateResult> postUserTemplateResult(UserTemplate userTemplate) {
    var request = mapper.toFacephiUserTemplateRequest(userTemplate);
    var response =
        webClient
            .post()
            .uri(Constant.POST_SERVICES_UPDATE_USER_TEMPLATE)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiUserTemplateResponse.class)
            .block();
    return buildWrapperExecution(
        mapper.toUserTemplateResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<IdentityResult> postIdentityResult(Identity identity) {
    var request = mapper.toFacephiIdentityRequest(identity);
    log.info("request: {}", request.toString());
    var response =
        webClient
            .post()
            .uri(Constant.POST_ONBOARDING_V2_IDENTITY)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiIdentityResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toIdentityResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<IadResult> postIadResult(Iad iad) {
    var request = iad.getResource();
    var response =
        webClient
            .post()
            .uri(Constant.POST_IAD)
            .contentType(MediaType.APPLICATION_OCTET_STREAM)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiIadResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toIadResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<ExtractDocumentDataResult> postExtractDocumentDataResult(
      ExtractDocumentData extractDocumentData) {
    var request = mapper.toFacephiExtractDocumentDataRequest(extractDocumentData);
    var response =
        webClient
            .post()
            .uri(Constant.POST_SERVICES_EXTRACT_DOCUMENT_DATA)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiExtractDocumentDataResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toExtractDocumentDataResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<ExtractDocumentDataResult> postExtractDocumentDataWebResult(
      ExtractDocumentDataWeb extractDocumentDataWeb) {
    var request = mapper.toFacephiExtractDocumentDataWebRequest(extractDocumentDataWeb);
    try {
      log.info("extract web request: {}", objectMapper.writeValueAsString(request));
    } catch (Exception e) {
      log.error("Error en writable : {}", e.getMessage());
    }

    var response =
        webClient
            .post()
            .uri(Constant.POST_SERVICES_EXTRACT_DOCUMENT_DATA_WEB)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiExtractDocumentDataResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toExtractDocumentDataResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<DocumentValidationStatusResult> postDocumentValidationStatusResult(
      DocumentValidationData documentValidationData) {
    var request = mapper.toFacephiDocumentValidationDataRequest(documentValidationData);
    var response =
        webClient
            .post()
            .uri(Constant.POST_VERIFY_DOCUMENT_VALIDATION_V2_STATUS)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiDocumentValidationStatusResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toDocumentValidationStatusResult(response),
        request,
        response,
        HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<DocumentValidationStatusResult> postDocumentValidationStartResult(
      DocumentValidationStart documentValidationStart) {
    var request = mapper.toFacephiDocumentValidationStartRequest(documentValidationStart);
    var response =
        webClient
            .post()
            .uri(Constant.POST_VERIFY_DOCUMENT_VALIDATION_V2_START)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiDocumentValidationStatusResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toDocumentValidationStatusResult(response),
        request,
        response,
        HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<DocumentValidationDataResult> postDocumentValidationDataResult(
      DocumentValidationData documentValidationData) {
    var request = mapper.toFacephiDocumentValidationDataRequest(documentValidationData);
    var response =
        webClient
            .post()
            .uri(Constant.POST_VERIFY_DOCUMENT_VALIDATION_V2_DATA)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiDocumentValidationDataResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toDocumentValidationDataResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<AuthenticateUserResult> postAuthenticateUserResult(
      AuthenticateUser authenticateUser) {
    var request = mapper.toFacephiAuthenticateUserRequest(authenticateUser);
    var response =
        webClient
            .post()
            .uri(Constant.POST_SERVICES_AUTHENTICATE_USER_V2)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiAuthenticateUserResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toAuthenticateUserResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<SignerResult> postSignerResult(Signer signer) {
    signer.setCallbackUrl(urlCallback);
    var request = mapper.toFacephiSignerRequest(signer);
    var response =
        webClient
            .post()
            .uri(Constant.POST_SIGNER)
            .bodyValue(request)
            .retrieve()
            .toBodilessEntity()
            .block();
    return buildWrapperExecution(
        mapper.toSignerResult(1, ""), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<CivilValidationResult> postCivilValidationResult(
      CivilValidation civilValidation) {
    var request = mapper.toFacephiCivilValidationRequest(civilValidation);
    var response =
        webClient
            .post()
            .uri(Constant.POST_CIVIL_VALIDATION)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiCivilValidationResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toCivilValidationResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<AuthenticateFacialResult> postAuthenticateFacialResult(
      AuthenticateFacial authenticateFacial) {
    var request = mapper.toFacephiAuthenticateFacialRequest(authenticateFacial);
    var response =
        webClient
            .mutate()
            .defaultHeader(Constant.X_API_KEY, apiProperties.getApiKeyAuth())
            .build()
            .post()
            .uri(Constant.POST_AUTHENTICATE_FACIAL)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiAuthenticateFacialResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toAuthenticateFacialResult(response), request, response, HttpStatus.OK.value());
  }

  @Override
  public WrapperExecution<EvaluatePassiveLivenessResult> postEvaluatePassiveLivenessResult(
      EvaluatePassiveLiveness evaluatePassiveLiveness) {
    var request = mapper.toFacephiEvaluatePassiveLivenessRequest(evaluatePassiveLiveness);
    var response =
        webClient
            .mutate()
            .defaultHeader(Constant.X_API_KEY, apiProperties.getApiKeyAuth())
            .build()
            .post()
            .uri(Constant.POST_EVALUATE_PASSIVE_LIVENESS)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(FacephiEvaluatePassiveLivenessResponse.class)
            .block();

    return buildWrapperExecution(
        mapper.toEvaluatePassiveLivenessResult(response), request, response, HttpStatus.OK.value());
  }

  private <T> WrapperExecution<T> buildWrapperExecution(
      T data, Object request, Object response, Integer httpStatus) {
    return WrapperExecution.<T>builder()
        .data(data)
        .request(toMap(request))
        .response(toMap(response))
        .httpStatus(httpStatus)
        .build();
  }

  private Map<String, Object> toMap(Object object) {
    if (object == null) {
      return Collections.emptyMap();
    }
    return objectMapper.convertValue(object, new TypeReference<>() {});
  }
}
